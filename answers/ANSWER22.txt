  this.gcRoots.add(id)
}

func markSweep() = {
  for (obj in this.gcHeap) {
    obj.set("alive", false)
  }
  for (rootId in this.gcRoots) {
    for (obj in this.gcHeap) {
      if (obj.get("id") === rootId) {
        obj.set("alive", true)
      }
    }
  }
  local swept = this.gcHeap.filter { obj -> obj.get("alive") }
  this.gcHeap = swept
  swept
}

allocate("c")

  local order = []
  func visit(m) = {

        for (d in deps) {
          visit(d)
        }
      }
      order.add(m)
    }

    visit(m)
  }
  order

  local edges = []
  local vars = live_ranges.keys()
  local n = vars.listLen()
  for (i: 1 .. n) {
    for (j: i+1 .. n) {
      local r1 = live_ranges.get(vars[i])

      local e1 = r1.get("end")

}

func buildDeadlockDetector(threads, locks) = {
  local graph = {}
  for (t in threads) {
    graph.set(t.get("id"), t.get("waiting_for"))
  }
  local visited = {}
  local rec_stack = {}
  local cycle = []
  func dfs(node) = {
    visited.set(node, true)
    rec_stack.set(node, true)
    local neighbor = graph.get(node)
    if (neighbor ? text) {
      if (!visited.containsKey(neighbor)) {
        dfs(neighbor)
      } else if (rec_stack.containsKey(neighbor)) {
        cycle.add(node _ " -> " _ neighbor)
      }
    }
    rec_stack.delete(node)
  }
  for (t in threads) {
    local tid = t.get("id")
    if (!visited.containsKey(tid)) {
      dfs(tid)
    }
  }
  if (cycle.listLen() > 0) cycle else ["no deadlock"]
}

  "E": [["T", "E'"]],
  "E'": [["+", "T", "E'"], [""]],
  "T": [["F", "T'"]],
  "T'": [["*", "F", "T'"], [""]],
  "F": [["(", "E", ")"], ["id"]]
}

local nonterminals = ["E", "E'", "T", "T'", "F"]
local firstSets = {}
for (nt in nonterminals) {
  firstSets.set(nt, [])
}

local changed = true
while (changed) {
  changed = false
  for (nt in nonterminals) {
    local prods = grammar.get(nt)
    for (prod in prods) {
      if (prod.listLen() == 1 && prod[1] === "") {
        local fs = firstSets.get(nt)
        if (!fs.containsItem("ε")) {
          fs.add("ε")
          changed = true
        }
      } else {
        local sym = prod[1]
        if (!nonterminals.containsItem(sym)) {
          local fs = firstSets.get(nt)
          if (!fs.containsItem(sym)) {
            fs.add(sym)
            changed = true
          }
        } else {
          local symFirst = firstSets.get(sym)
          for (t in symFirst) {
            if (t !== "ε") {
              local fs = firstSets.get(nt)
              if (!fs.containsItem(t)) {
                fs.add(t)
                changed = true
              }
            }
          }
        }
      }
    }
  }

  "E+id": ["T", "E'"],
  "E+(": ["T", "E'"],
  "E'++": ["+", "T", "E'"],
  "E'+)": [],
  "E'+$": [],
  "T+id": ["F", "T'"],
  "T+(": ["F", "T'"],
  "T'+*": ["*", "F", "T'"],
  "T'++": [],
  "T'+)": [],
  "T'+$": [],
  "F+id": ["id"],
  "F+(": ["(", "E", ")"]
}

local input_ll1 = ["id", "+", "id", "*", "id", "$"]
local stack_ll1 = ["$", "E"]
local pos_ll1 = 1
local accepted = false
local error_ll1 = false

while (stack_ll1.listLen() > 0 && !accepted && !error_ll1) {
  local top = stack_ll1[stack_ll1.listLen()]
  local curr = input_ll1[pos_ll1]
  println("Stack top: " _ top _ " Input: " _ curr)
  if (top === "$" && curr === "$") {
    accepted = true
  } else if (top === curr) {
    stack_ll1.remove(stack_ll1.listLen())
    pos_ll1 = pos_ll1 + 1
  } else {
    local key = top _ "+" _ curr
    if (ll1_table.containsKey(key)) {
      local prod = ll1_table.get(key)
      stack_ll1.remove(stack_ll1.listLen())
      for (i: prod.listLen() .. 1 step -1) {
        if (prod[i] !== "") {
          stack_ll1.add(prod[i])
        }
      }
    } else {
      error_ll1 = true
    }
  }
}
println(if (accepted) "Accepted!" else "Parse error")

func deinterleave_bits(z) = {
  local x = 0
  local y = 0
  local bit = 0
  local zz = z
  while (zz > 0) {
    x = x + (zz % 2) * (2 ^ bit)
    zz = floor(zz / 2)
    y = y + (zz % 2) * (2 ^ bit)
    zz = floor(zz / 2)
    bit = bit + 1
  }
  {"x": x, "y": y}
}

func interleave_bits(x, y) = {
  local z = 0
  local xx = x
  local yy = y
  local bit = 0
  while (xx > 0 || yy > 0) {
    z = z + (xx % 2) * (2 ^ (2*bit))
    z = z + (yy % 2) * (2 ^ (2*bit+1))
    xx = floor(xx / 2)
    yy = floor(yy / 2)
    bit = bit + 1
  }
  z
}

func karatsuba(a, b) = {
  if (a < 10 || b < 10) {
    a * b
  } else {
    local n = max(floor(log(a)/log(10)) + 1, floor(log(b)/log(10)) + 1)
    local m = floor(n / 2)
    local p = 10 ^ m
    local a1 = floor(a / p)
    local a0 = a % p
    local b1 = floor(b / p)
    local b0 = b % p
    local z0 = karatsuba(a0, b0)
    local z2 = karatsuba(a1, b1)
    local z1 = karatsuba(a0 + a1, b0 + b1) - z2 - z0
    z2 * p * p + z1 * p + z0
  }
}

println(karatsuba(1234, 5678))

func companion_matrix(poly) = {
  local n = poly.listLen() - 1
  local C = []
  for (i: 1 .. n) {
    local row = []
    for (j: 1 .. n) {
      row.add(0)
    }
    C.add(row)
  }
  for (i: 1 .. n-1) {
    C[i+1][i] = 1
  }
  local lead = poly[n+1]
  for (j: 1 .. n) {
    C[j][n] = -poly[j] / lead
  }
  C
}

func computeLiveDefs(blocks) = {
  local result = {}
  for (b in blocks) {
    local def_set = []
    local use_set = []
    local instrs = b.get("instrs")
    for (instr in instrs) {
      if (instr.containsKey("def") && !def_set.containsItem(instr.get("def"))) {
        def_set.add(instr.get("def"))

      if (instr.containsKey("use")) {
        for (u in instr.get("use")) {
          if (!def_set.containsItem(u) && !use_set.containsItem(u)) {
            use_set.add(u)

        }
      }
    }

  }

func modulo_schedule(ops, latencies, resources) = {
  local n = ops.listLen()
  local max_lat = latencies.max { m, n -> m > n }
  local ii = max_lat
  local schedule = []
  for (cycle: 0 .. ii - 1) {
    local slot = []
    for (i: 1 .. n) {
      if ((i - 1) % ii == cycle) {
        slot.add({"op": ops[i], "latency": latencies[i], "cycle": cycle})
      }
    }
    schedule.add(slot)
  }
  schedule
}

    local op = instr.get("op")

      interp.get("stack").add(instr.get("val"))
    } else if (op === "add") {
      local b = interp.get("stack")[interp.get("stack").listLen()]
      interp.get("stack").remove(interp.get("stack").listLen())
      local a = interp.get("stack")[interp.get("stack").listLen()]
      interp.get("stack").remove(interp.get("stack").listLen())
      interp.get("stack").add(a + b)
    } else if (op === "print") {
      local val = interp.get("stack")[interp.get("stack").listLen()]
      interp.get("output").add(val)
    }
  }
  interp
}

func typeCheck(ast, env) = {
  local errors = []
  local typed_ast = copyDict(ast)
  func checkNode(node) = {
    local kind = node.get("kind")
    if (kind === "var") {
      local name = node.get("name")
      if (!env.containsKey(name)) {
        errors.add("Undefined variable: " _ name)
        "unknown"
      } else {
        env.get(name)
      }
    } else if (kind === "num") {
      "number"
    } else if (kind === "binop") {
      local lt = checkNode(node.get("left"))
      local rt = checkNode(node.get("right"))
      if (lt !== "number" || rt !== "number") {
        errors.add("Type error in binop: expected number, got " _ lt _ " and " _ rt)
      }
      "number"
    } else {
      "unknown"
    }
  }
  local result_type = checkNode(ast)
  typed_ast.set("inferred_type", result_type)
  {"typed_ast": typed_ast, "errors": errors}
}

func formatSource(fmt, source) = {
  local lines = source.split("\n")
  local result = ""
  local level = 0
  for (line in lines) {
    local trimmed = line.trim()
    if (trimmed.endsWith("}")) {
      level = if (level > 0) level - 1 else 0
    }
    local indent = ""
    for (i: 1 .. level) {
      indent = indent _ fmt.get("indent")
    }
    result = result _ indent _ trimmed _ "\n"
    if (trimmed.endsWith("{")) {
      level = level + 1
    }
  }
  result
}

func matchAST(matcher, ast) = {
  local pat = matcher.get("pattern")
  func matchNode(p, n) = {
    if (p.get("kind") !== n.get("kind")) {
      false
    } else if (p.containsKey("name") && p.get("name") !== "_" && p.get("name") !== n.get("name")) {
      false
    } else {
      true
    }
  }
  func walk(node) = {
    local matches = []
    if (matchNode(pat, node)) {
      matches.add(node)
    }
    if (node.containsKey("children")) {
      for (child in node.get("children")) {
        matches.appendList(walk(child))
      }
    }
    matches
  }
  walk(ast)
}

func buildRaceDetector(program, threads) = {
  local accesses = {}
  local races = []
  for (thread in threads) {
    local tid = thread.get("id")
    for (access in thread.get("accesses")) {
      local addr = access.get("addr")
      local op = access.get("op")
      if (!accesses.containsKey(addr)) {
        accesses.set(addr, [])
      }
      accesses.get(addr).add({"tid": tid, "op": op})
    }
  }
  local addrs = accesses.keys()
  for (addr in addrs) {
    local acc_list = accesses.get(addr)
    local has_write = acc_list.filter { a -> a.get("op") === "write" }.listLen() > 0
    local multi_thread = acc_list.listLen() > 1
    if (has_write && multi_thread) {
      races.add({"addr": addr, "accesses": acc_list})
    }
  }
  races
}

func parse(tokens) = {
  local pos = [1]
  func peek() = {
    if (pos[1] <= tokens.listLen()) tokens[pos[1]] else {"type":"EOF","value":""}
  }
  func consume() = {
    local tok = peek()
    pos[1] = pos[1] + 1
    tok
  }
  func parseExprP() = {
    local left = {"kind":"var","name":consume().get("value")}
    if (peek().get("type") === "OP") {
      local op = consume().get("value")
      local right = {"kind":"var","name":consume().get("value")}
      {"kind":"binop","op":op,"left":left,"right":right}
    } else {
      left
    }
  }
  func parseStmt() = {
    local tok = peek()
    if (tok.get("value") === "if") {
      consume()
      consume()
      local cond = parseExprP()
      consume()
      consume()
      local then_expr = {"kind":"return","val":parseExprP()}
      consume()
      consume()
      local else_expr = {"kind":"return","val":parseExprP()}
      consume()
      {"kind":"if","cond":cond,"then":then_expr,"else":else_expr}
    } else {
      {"kind":"unknown"}
    }
  }
  parseStmt()
}

local parse_tokens = [
  {"type":"KEYWORD","value":"if"},
  {"type":"OP","value":"("},
  {"type":"IDENT","value":"x"},
  {"type":"OP","value":">"},
  {"type":"NUMBER","value":"0"},
  {"type":"OP","value":")"},
  {"type":"KEYWORD","value":"return"},
  {"type":"IDENT","value":"x"},
  {"type":"OP","value":";"},
  {"type":"KEYWORD","value":"else"},
  {"type":"KEYWORD","value":"return"},
  {"type":"OP","value":"-"},
  {"type":"IDENT","value":"x"},
  {"type":"OP","value":";"}
]
