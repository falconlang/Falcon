// 1770
func giniCoefficient(incomes) = {
  local sorted = copyList(incomes).sort { a, b -> a < b }
  local n = sorted.listLen()
  if (n == 0) 0 else {
    local sumAll = sorted.reduce(0) { acc, v -> acc + v }
    local runSum = 0
    local giniSum = 0
    for (i: 1 .. n) {
      runSum = runSum + sorted[i]
      local shareY = runSum / sumAll
      local shareX = i / n
      giniSum = giniSum + (shareX - shareY)
    }
    (giniSum / n) * 2
  }
}

// 1845
func levenshtein(s1, s2) = {
  local dp =[]
  for (i: 0 .. s1.textLen()) {
    local row =[]
    for (j: 0 .. s2.textLen()) { row.add(0) }
    dp.add(row)
  }
  for (i: 1 .. s1.textLen() + 1) { dp[i][1] = i - 1 }
  for (j: 1 .. s2.textLen() + 1) { dp[1][j] = j - 1 }
  for (i: 2 .. s1.textLen() + 1) {
    for (j: 2 .. s2.textLen() + 1) {
      local cost = if (s1.segment(i - 1, 1) === s2.segment(j - 1, 1)) 0 else 1
      local del = dp[i - 1][j] + 1
      local ins = dp[i][j - 1] + 1
      local sub = dp[i - 1][j - 1] + cost
      dp[i][j] = min(del, min(ins, sub))
    }
  }
  dp[s1.textLen() + 1][s2.textLen() + 1]
}

func bucketSort(list, B) = {

  for (x in list) {
    local idx = floor(x * B) + 1
    if (idx > B) idx = B
    buckets[idx].add(x)
  }
  local res =[]
  for (b in buckets) {
    local sortedB = b.sort { m, n -> m < n }
    for (val in sortedB) res.add(val)
  }
  res
}

        local v = r.get(k)
        if (v < limits.get("min") || v > limits.get("max")) violations.add({"row": i, "error": "Range violation " _ k})
      }
    }
  }
  violations
}

func bfsShortestPath(graph, start, endNode) = {
  local q =[[start]]
  local visited =[start]
  local path =[]
  while (q.listLen() > 0) {
    local currPath = q[1]
    q.remove(1)
    local node = currPath[currPath.listLen()]
    if (node === endNode) { path = currPath break }
    for (neighbor in graph.get(node,[])) {
      if (!visited.containsItem(neighbor)) {
        visited.add(neighbor)
        local np = copyList(currPath)
        np.add(neighbor)
        q.add(np)
      }
    }
  }
  path
}

// 1761
func simpleLinearRegression(xList, yList) = {
  local n = xList.listLen()
  local sx = 0
  local sy = 0
  local sxy = 0
  local sxx = 0
  for (i: 1 .. n) {
    sx = sx + xList[i]
    sy = sy + yList[i]
    sxy = sxy + xList[i] * yList[i]
    sxx = sxx + xList[i] * xList[i]
  }
  local m = (n * sxy - sx * sy) / (n * sxx - sx * sx)
  local b = (sy - m * sx) / n
  {"slope": m, "intercept": b}
}

    local right = mergeSortHelper(list.slice(mid + 1, list.listLen()))
    local res =[]
    local i = 1
    local j = 1
    while (i <= left.listLen() && j <= right.listLen()) {
      if (left[i] < right[j]) { res.add(left[i]); i = i + 1 }
      else { res.add(right[j]); j = j + 1 }
    }
    while (i <= left.listLen()) { res.add(left[i]); i = i + 1 }
    while (j <= right.listLen()) { res.add(right[j]); j = j + 1 }
    res
  }
}

  local tLen = 0
  local bounces = 0
  local entries = {}
  for (s in sessions) {
    tLen = tLen + s.listLen()
    if (s.listLen() == 1) bounces = bounces + 1
    local ep = s[1].get("page")
    entries.set(ep, entries.get(ep, 0) + 1)
  }
  local bestE = ""
  local mE = 0
  for (k, v in entries) { if (v > mE) { mE = v bestE = k } }
  {"avgLen": tLen / sessions.listLen(), "bounceRate": bounces / sessions.listLen(), "topEntry": bestE}
}

func markovSteadyState(trans) = {
  local counts = {}
  local state = trans.keys()[1]
  for (i: 1 .. 10000) {
    counts.set(state, counts.get(state, 0) + 1)
    state = trans.get(state).keys()[1] // Simple mocked transition
  }
  counts
}

// 1834
func simpleTokenizer(sentence) = {
  local words = sentence.splitAtSpaces()
  local res =[]
  for (w in words) {
    local lw = w.lowercase()
    if (!(lw.textLen() == 1 && lw === ".")) res.add(lw) // Simplistic check
  }
  res
}

// 1850
func parseJSONKeyValue(texts) = {
  local res = {}
  for (t in texts) {
    local p = t.split(":")
    res.set(p[1].replace("\"", ""), p[2].replace("\"", ""))
  }
  res
}

// 2080
func categoryColorHash(categories) = {
  local mapping = {}
  for (c in categories) {
    local h = mod(c.textLen() * 13 + c.segment(1,1).textLen(), 256)
    mapping.set(c, makeColor([mod(h*31, 256), mod(h*57, 256), mod(h*97, 256)]))
  }
  mapping
}
