// 6609
func wcagLuminance(color) = {
  local c = splitColor(color)
  0.2126*(c[2]/255) + 0.7152*(c[3]/255) + 0.0722*(c[4]/255)
}
local fg = makeColor([0, 0, 0])
local bg = makeColor([255, 255, 255])
local L1 = wcagLuminance(bg)
local L2 = wcagLuminance(fg)
local ratio = (L1+0.05)/(L2+0.05)
println("Contrast ratio: " _ formatDecimal(ratio, 2))
println(if (ratio >= 4.5) "WCAG AA passed" else "WCAG AA failed")
