// 199
func interpolate(points, x) = {
  // Assuming list of [x, y] sorted by x
  local pt1 = points[1]
  local pt2 = points[points.listLen()]
  for (i: 1 .. points.listLen() - 1) {
    if (points[i][1] <= x && points[i+1][1] >= x) {
      pt1 = points[i]
      pt2 = points[i+1]
      break
    }
  }
  local t = if (pt2[1] == pt1[1]) 0 else (x - pt1[1]) / (pt2[1] - pt1[1])
  pt1[2] + (pt2[2] - pt1[2]) * t
}

// 375
func rotateChars(s, n) = {
  local alpha = "abcdefghijklmnopqrstuvwxyz"
  local res = ""
  for (i: 1 .. s.textLen()) {
    local ch = s.segment(i, 1)
    local lower = ch.lowercase()
    local idx = -1
    for(j: 1 .. 26) { if(alpha.segment(j, 1) === lower) idx = j }
    if (idx == -1) {
      res = res _ ch
    } else {
      local newCh = alpha.segment((idx + n - 1) % 26 + 1, 1)
      res = res _ if (ch === lower) newCh else newCh.uppercase()
    }
  }
  res
}

func nQueens(n) = {
  func solve(row, cols, diag1, diag2) = {
    if(row > n) 1 else {
      local count = 0
      for(c: 1 .. n) {
        if(!cols.containsItem(c) && !diag1.containsItem(row - c) && !diag2.containsItem(row + c)) {
          cols.add(c) diag1.add(row - c) diag2.add(row + c)
          count = count + solve(row + 1, cols, diag1, diag2)
          cols.remove(cols.listLen()) diag1.remove(diag1.listLen()) diag2.remove(diag2.listLen())
        }
      }
      count
    }
  }
  solve(1, [], [],[])
}

// 957
func knapsack(wt, val, cap) = {
  local n = wt.listLen()
  local dp =[]
  for(i: 1 .. cap + 1) { dp.add(0) }
  for(i: 1 .. n) {
    for(w: cap + 1 .. 1 step -1) {
      if (wt[i] <= w - 1) {
        local test = dp[w - wt[i]] + val[i]
        if(test > dp[w]) dp[w] = test
      }
    }
  }
  dp[cap + 1]
}

func topologicalSort(graph) = {
  local inD = {}
  for(u in graph.keys()) {
    if(!inD.containsKey(u)) inD.set(u, 0)
    for(v in graph.get(u)) { inD.set(v, inD.getAtPath([v], 0) + 1) }
  }
  local q =[]
  for(k, v in inD) { if(v == 0) q.add(k) }
  local res =[]
  while(q.listLen() > 0) {
    local u = q[1]
    q.remove(1)
    res.add(u)
    if(graph.containsKey(u)) {
      for(v in graph.get(u)) {
        inD.set(v, inD.get(v) - 1)
        if(inD.get(v) == 0) q.add(v)
      }
    }
  }
  res
}
