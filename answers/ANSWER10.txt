  local n = s2.textLen()

    local row = []
    for (j: 0 .. n) { row.add(0) }
    dp.add(row)
  }
  for (i: 1 .. m) {
    for (j: 1 .. n) {
      if (s1.segment(i,1) === s2.segment(j,1)) {
        dp[i][j] = dp[i-1][j-1] + 1
      } else {
        dp[i][j] = if (dp[i-1][j] > dp[i][j-1]) dp[i-1][j] else dp[i][j-1]
      }
    }
  }
  dp[m][n]
}

func mergeCount(arr) = {
  if (arr.listLen() <= 1) {
    [arr, 0]
  } else {
    local mid = floor(arr.listLen() / 2)
    local left = arr.slice(1, mid)
    local right = arr.slice(mid+1, arr.listLen())
    local lr = mergeCount(left)
    local rr = mergeCount(right)
    local lSorted = lr[1]
    local rSorted = rr[1]
    local inv = lr[2] + rr[2]
    local merged = []
    local i = 1
    local j = 1
    while (i <= lSorted.listLen() && j <= rSorted.listLen()) {
      if (lSorted[i] <= rSorted[j]) {
        merged.add(lSorted[i])
        i = i + 1
      } else {
        merged.add(rSorted[j])
        inv = inv + lSorted.listLen() - i + 1
        j = j + 1
      }
    }
    while (i <= lSorted.listLen()) { merged.add(lSorted[i]) i = i+1 }
    while (j <= rSorted.listLen()) { merged.add(rSorted[j]) j = j+1 }
    [merged, inv]
  }
}

func kirchhoff(adj, n) = {
  local L = []
  for (i: 1 .. n) {
    local row = []
    for (j: 1 .. n) { row.add(0) }
    L.add(row)
  }
  for (i: 1 .. n) {
    for (j: 1 .. n) {
      if (adj[i][j] == 1) {
        L[i][i] = L[i][i] + 1
        L[i][j] = L[i][j] - 1
      }
    }
  }
  local M = []
  for (i: 1 .. n-1) {
    local row = []
    for (j: 1 .. n-1) { row.add(L[i][j]) }
    M.add(row)
  }
  local det = 0
  if (M.listLen() == 2) {
    det = M[1][1]*M[2][2] - M[1][2]*M[2][1]
  } else if (M.listLen() == 3) {
    det = M[1][1]*(M[2][2]*M[3][3]-M[2][3]*M[3][2]) - M[1][2]*(M[2][1]*M[3][3]-M[2][3]*M[3][1]) + M[1][3]*(M[2][1]*M[3][2]-M[2][2]*M[3][1])
  } else {
    det = M[1][1]
  }
  round(abs(det))
}

  else {
    local factors = []

    local d = 2
    local squareFree = true

        if (x%d==0) { squareFree = false }
      }
      d = d+1
    }

    if (!squareFree) { 0 } else { (-1)^factors.listLen() }
  }
}

func eulerPhi(n) = {
  local result = n
  local x = n
  local d = 2
  while (d*d <= x) {
    if (x%d==0) {
      while (x%d==0) { x = floor(x/d) }
      result = result - floor(result/d)
    }
    d = d + 1
  }
  if (x > 1) { result = result - floor(result/x) }
  result
}
