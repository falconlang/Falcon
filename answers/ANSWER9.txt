func grahamScan(points) = {
  local pivot = points[1]
  for (p in points) {
    if (p[2] < pivot[2] || (p[2] == pivot[2] && p[1] < pivot[1])) {
      pivot = p
    }
  }
  local sorted = copyList(points)
  sorted.sort { a, b ->
    local ax = a[1] - pivot[1]
    local ay = a[2] - pivot[2]
    local bx = b[1] - pivot[1]
    local by = b[2] - pivot[2]
    local cross = ax * by - ay * bx
    cross > 0
  }
  local hull = [sorted[1], sorted[2]]
  for (i: 3 .. sorted.listLen()) {
    while (hull.listLen() > 1) {
      local a = hull[hull.listLen()-1]
      local b = hull[hull.listLen()]
      local c = sorted[i]
      local cross = (b[1]-a[1])*(c[2]-a[2]) - (b[2]-a[2])*(c[1]-a[1])
      if (cross <= 0) { hull.remove(hull.listLen()) } else { a = hull[1] }
    }
    hull.add(sorted[i])
  }
  hull
}
println(grahamScan([[0,0],[1,1],[2,0],[1,2],[0,2],[2,2]]))

func astarSearch(graph, start, goal, heuristic) = {
  local openSet = [[heuristic.get(start,0), 0, start, [start]]]
  local visited = []
  local result = []
  while (openSet.listLen() > 0) {
    openSet.sort()
    local cur = openSet[1]
    openSet.remove(1)
    local f = cur[1]
    local g = cur[2]
    local node = cur[3]
    local path = cur[4]
    if (node === goal) {
      result = path
      openSet = []
    } else if (!visited.containsItem(node)) {
      visited.add(node)
      for (nb, w in graph.get(node, {})) {
        if (!visited.containsItem(nb)) {
          local ng = g + w
          local nf = ng + heuristic.get(nb, 0)
          local np = copyList(path)
          np.add(nb)
          openSet.add([nf, ng, nb, np])
        }
      }
    }
  }
  result
}

func normalizePoints(points) = {
  local minX = points[1][1]
  local minY = points[1][2]
  local maxX = points[1][1]
  local maxY = points[1][2]
  for (p in points) {
    if (p[1] < minX) { minX = p[1] }
    if (p[2] < minY) { minY = p[2] }
    if (p[1] > maxX) { maxX = p[1] }
    if (p[2] > maxY) { maxY = p[2] }
  }
  local rx = if (maxX == minX) 1 else maxX - minX
  local ry = if (maxY == minY) 1 else maxY - minY
  points.map { p -> [(p[1]-minX)/rx, (p[2]-minY)/ry] }
}

func computeSDF(polygons, queryPt) = {
  local minDist = 999999
  local inside = false
  for (poly in polygons) {
    local n = poly.listLen()
    local isInside = false
    local px = queryPt[1]
    local py = queryPt[2]
    local j = n
    for (i: 1 .. n) {
      local xi = poly[i][1]
      local yi = poly[i][2]
      local xj = poly[j][1]
      local yj = poly[j][2]
      if (((yi > py) != (yj > py)) && px < (xj-xi)*(py-yi)/(yj-yi)+xi) {
        isInside = !isInside
      }
      local dx = poly[i][1] - px
      local dy = poly[i][2] - py
      local d = sqrt(dx*dx+dy*dy)
      if (d < minDist) { minDist = d }
      j = i
    }
    if (isInside) { inside = true }
  }
  if (inside) { -minDist } else { minDist }
}

    }

func hausdorff(setA, setB) = {
  local maxMinA = 0
  for (a in setA) {
    local minD = 999999
    for (b in setB) {
      local dx = a[1]-b[1]
      local dy = a[2]-b[2]
      local d = sqrt(dx*dx+dy*dy)
      if (d < minD) { minD = d }
    }
    if (minD > maxMinA) { maxMinA = minD }
  }
  local maxMinB = 0
  for (b in setB) {
    local minD = 999999
    for (a in setA) {
      local dx = a[1]-b[1]
      local dy = a[2]-b[2]
      local d = sqrt(dx*dx+dy*dy)
      if (d < minD) { minD = d }
    }
    if (minD > maxMinB) { maxMinB = minD }
  }
  if (maxMinA > maxMinB) { maxMinA } else { maxMinB }
}

        local dy = pt[2]-uy
        local r2 = (ax-ux)*(ax-ux)+(ay-uy)*(ay-uy)
        if (dx*dx+dy*dy < r2) { badTris.add(tri) } else { goodTris.add(tri) }
      }
    }

    for (bt in badTris) {
      local edges = [[bt[1],bt[2]],[bt[2],bt[3]],[bt[3],bt[1]]]
      for (e in edges) {
        local shared = false
        for (bt2 in badTris) {

            }
          }
        }
        if (!shared) { boundary.add(e) }
      }
    }
    triangles = goodTris
    for (e in boundary) {
      triangles.add([e[1], e[2], pi])
    }
  }
  local finalTris = []
  for (tri in triangles) {
    if (tri[1] <= 3 || tri[2] <= 3 || tri[3] <= 3) {
    } else {
      finalTris.add([tri[1]-3, tri[2]-3, tri[3]-3])
    }
  }

}

func findTIntersections(segments) = {

  for (i: 1 .. segments.listLen()) {
    for (j: 1 .. segments.listLen()) {
      if (i != j) {
        local si = segments[i]
        local sj = segments[j]
        local endpoints = [si[1], si[2]]
        for (ep in endpoints) {
          local ax = sj[1][1]
          local ay = sj[1][2]
          local bx = sj[2][1]
          local by = sj[2][2]
          local dx = bx-ax
          local dy = by-ay
          local len2 = dx*dx+dy*dy
          if (len2 > 0) {
            local t = ((ep[1]-ax)*dx+(ep[2]-ay)*dy)/len2
            if (t > 0.001 && t < 0.999) {
              local px = ax+t*dx
              local py = ay+t*dy
              local ex = ep[1]-px
              local ey = ep[2]-py
              if (ex*ex+ey*ey < 0.0001) {
                result.add({"point":ep,"segment":j,"t":t})
              }
            }
          }
        }
      }
    }
  }
  result
}

func welzl(pts, boundary) = {
  if (pts.listLen() == 0 || boundary.listLen() == 3) {
    if (boundary.listLen() == 0) { {"cx":0,"cy":0,"r":0} }
    else if (boundary.listLen() == 1) { {"cx":boundary[1][1],"cy":boundary[1][2],"r":0} }
    else if (boundary.listLen() == 2) {
      local mx = (boundary[1][1]+boundary[2][1])/2
      local my = (boundary[1][2]+boundary[2][2])/2
      local dx = boundary[1][1]-boundary[2][1]
      local dy = boundary[1][2]-boundary[2][2]
      {"cx":mx,"cy":my,"r":sqrt(dx*dx+dy*dy)/2}
    } else {
      local ax=boundary[1][1] local ay=boundary[1][2]
      local bx=boundary[2][1] local by=boundary[2][2]
      local cx=boundary[3][1] local cy=boundary[3][2]
      local D = 2*(ax*(by-cy)+bx*(cy-ay)+cx*(ay-by))
      local ux = ((ax*ax+ay*ay)*(by-cy)+(bx*bx+by*by)*(cy-ay)+(cx*cx+cy*cy)*(ay-by))/D
      local uy = ((ax*ax+ay*ay)*(cx-bx)+(bx*bx+by*by)*(ax-cx)+(cx*cx+cy*cy)*(bx-ax))/D
      local r = sqrt((ax-ux)*(ax-ux)+(ay-uy)*(ay-uy))
      {"cx":ux,"cy":uy,"r":r}
    }
  } else {
    local p = pts[pts.listLen()]
    local rest = pts.slice(1, pts.listLen()-1)
    local circle = welzl(rest, boundary)
    local dx = p[1]-circle.get("cx",0)
    local dy = p[2]-circle.get("cy",0)
    if (dx*dx+dy*dy <= circle.get("r",0)^2 + 0.0001) {
      circle
    } else {
      local nb = copyList(boundary)
      nb.add(p)
      welzl(rest, nb)
    }
  }
}

    local len2 = sqrt(dx2*dx2+dy2*dy2)

  }

}

        local d = dx*dx+dy*dy
        if (d > maxDist) {
          maxDist = d
          pair = [hull[i], hull[j]]
        }
        j = nj
        j = if (j >= n) n else j
        j = n
      }
    }
  }
  [pair, sqrt(maxDist)]
}
local convexHull = [[0,0],[2,0],[3,2],[2,4],[0,4],[-1,2]]
println(farthestPair(convexHull))

func bresenham(x1, y1, x2, y2) = {
  local pts = []
  local dx = abs(x2-x1)
  local dy = abs(y2-y1)
  local sx = if (x1 < x2) 1 else -1
  local sy = if (y1 < y2) 1 else -1
  local err = dx - dy
  local x = x1
  local y = y1
  while (true) {
    pts.add([x, y])
    if (x == x2 && y == y2) { x = x2 + 1 }
    else {
      local e2 = 2 * err
      if (e2 > -dy) { err = err - dy x = x + sx }
      if (e2 < dx) { err = err + dx y = y + sy }
    }
  }
  pts
}
println(bresenham(0, 0, 5, 3))

func sutherlandHodgman(poly, clipPoly) = {
  local output = copyList(poly)
  local cn = clipPoly.listLen()
  for (i: 1 .. cn) {
    local input = copyList(output)
    output = []
    local edgeA = clipPoly[i]
    local edgeB = clipPoly[if (i==cn) 1 else i+1]
    for (j: 1 .. input.listLen()) {
      local cur = input[j]
      local prev = input[if (j==1) input.listLen() else j-1]
      local ex = edgeB[1]-edgeA[1]
      local ey = edgeB[2]-edgeA[2]
      local curInside = ex*(cur[2]-edgeA[2])-ey*(cur[1]-edgeA[1]) >= 0
      local prevInside = ex*(prev[2]-edgeA[2])-ey*(prev[1]-edgeA[1]) >= 0
      if (curInside) {
        if (!prevInside) {
          local dx = cur[1]-prev[1]
          local dy = cur[2]-prev[2]
          local t = (ex*(prev[2]-edgeA[2])-ey*(prev[1]-edgeA[1])) / (ey*dx-ex*dy)
          output.add([prev[1]+t*dx, prev[2]+t*dy])
        }
        output.add(cur)
      } else if (prevInside) {
        local dx = cur[1]-prev[1]
        local dy = cur[2]-prev[2]
        local t = (ex*(prev[2]-edgeA[2])-ey*(prev[1]-edgeA[1])) / (ey*dx-ex*dy)
        output.add([prev[1]+t*dx, prev[2]+t*dy])
      }
    }
  }
  local area = 0
  local n = output.listLen()
  for (i: 1 .. n) {
    local j = if (i==n) 1 else i+1
    area = area + output[i][1]*output[j][2] - output[j][1]*output[i][2]
  }
  abs(area)/2
}
println(sutherlandHodgman([[0,0],[4,0],[4,4],[0,4]], [[1,1],[3,1],[3,3],[1,3]]))

func douglasPeucker(pts, epsilon) = {
  if (pts.listLen() <= 2) {
    pts
  } else {
    local ax = pts[1][1]
    local ay = pts[1][2]
    local bx = pts[pts.listLen()][1]
    local by = pts[pts.listLen()][2]
    local maxDist = 0
    local maxIdx = 1
    for (i: 2 .. pts.listLen()-1) {
      local dx = bx-ax
      local dy = by-ay
      local len = sqrt(dx*dx+dy*dy)
      local d = if (len == 0) 0 else abs((pts[i][1]-ax)*dy-(pts[i][2]-ay)*dx)/len
      if (d > maxDist) { maxDist = d maxIdx = i }
    }
    if (maxDist > epsilon) {
      local left = douglasPeucker(pts.slice(1, maxIdx), epsilon)
      local right = douglasPeucker(pts.slice(maxIdx, pts.listLen()), epsilon)
      local result = copyList(left)
      for (i: 2 .. right.listLen()) { result.add(right[i]) }
      result
    } else {
      [pts[1], pts[pts.listLen()]]
    }
  }
}
println(douglasPeucker([[0,0],[1,0.1],[2,0],[3,0.5],[4,0],[5,0]], 0.3))

func buildBVH(objects) = {
  if (objects.listLen() <= 1) {
    {"type":"leaf","objects":objects}
  } else {
    local minX=999999 local maxX=-999999 local minY=999999 local maxY=-999999
    for (o in objects) {
      local b = o.get("bbox",[0,0,1,1])
      if (b[1]<minX){minX=b[1]} if (b[3]>maxX){maxX=b[3]}
      if (b[2]<minY){minY=b[2]} if (b[4]>maxY){maxY=b[4]}
    }
    local splitAxis = if (maxX-minX > maxY-minY) "x" else "y"
    local sorted = copyList(objects)
    sorted.sort { a, b ->
      local ba = a.get("bbox",[0,0,0,0])
      local bb = b.get("bbox",[0,0,0,0])
      if (splitAxis === "x") (ba[1]+ba[3])/2 < (bb[1]+bb[3])/2
      else (ba[2]+ba[4])/2 < (bb[2]+bb[4])/2
    }
    local mid = floor(sorted.listLen()/2)
    {"type":"node","bbox":[minX,minY,maxX,maxY],"left":buildBVH(sorted.slice(1,mid)),"right":buildBVH(sorted.slice(mid+1,sorted.listLen()))}
  }
}
local objs = [{"id":1,"bbox":[0,0,2,2]},{"id":2,"bbox":[3,3,5,5]},{"id":3,"bbox":[1,4,2,6]},{"id":4,"bbox":[4,0,6,2]}]
println(buildBVH(objs))

func subdivideTriangle(a, b, c, depth) = {
  if (depth == 0) {
    [[a, b, c]]
  } else {
    local mab = [(a[1]+b[1])/2, (a[2]+b[2])/2]
    local mbc = [(b[1]+c[1])/2, (b[2]+c[2])/2]
    local mac = [(a[1]+c[1])/2, (a[2]+c[2])/2]
    local result = []
    for (t in subdivideTriangle(a, mab, mac, depth-1)) { result.add(t) }
    for (t in subdivideTriangle(mab, b, mbc, depth-1)) { result.add(t) }
    for (t in subdivideTriangle(mac, mbc, c, depth-1)) { result.add(t) }
    for (t in subdivideTriangle(mab, mbc, mac, depth-1)) { result.add(t) }
    result
  }
}
local tris = subdivideTriangle([0,0],[4,0],[2,4], 2)
println(tris.listLen())

func fitPlane(pts) = {
  local n = pts.listLen()
  local cx = 0
  local cy = 0
  local cz = 0
  for (p in pts) { cx=cx+p[1] cy=cy+p[2] cz=cz+p[3] }
  cx=cx/n cy=cy/n cz=cz/n
  local xx=0 local xy=0 local xz=0 local yy=0 local yz=0 local zz=0
  for (p in pts) {
    local rx=p[1]-cx local ry=p[2]-cy local rz=p[3]-cz
    xx=xx+rx*rx xy=xy+rx*ry xz=xz+rx*rz yy=yy+ry*ry yz=yz+ry*rz zz=zz+rz*rz
  }
  local nx = yy*zz-yz*yz
  local ny = xz*yz-xy*zz
  local nz = xy*yz-yy*xz
  local len = sqrt(nx*nx+ny*ny+nz*nz)
  if (len > 0) { {"normal":[nx/len,ny/len,nz/len],"centroid":[cx,cy,cz]} }
  else { {"normal":[0,0,1],"centroid":[cx,cy,cz]} }
}
println(fitPlane([[0,0,0],[1,0,0],[0,1,0],[1,1,0.1],[0.5,0.5,0.05]]))

func closestPointOnSegment(pt, segA, segB) = {
  local dx = segB[1]-segA[1]
  local dy = segB[2]-segA[2]
  local len2 = dx*dx+dy*dy
  if (len2 == 0) {
    segA
  } else {
    local t = ((pt[1]-segA[1])*dx+(pt[2]-segA[2])*dy)/len2
    local tc = if (t<0) 0 else if (t>1) 1 else t
    [segA[1]+tc*dx, segA[2]+tc*dy]
  }
}
println(closestPointOnSegment([3,4],[0,0],[5,0]))
println(closestPointOnSegment([6,4],[0,0],[5,0]))
