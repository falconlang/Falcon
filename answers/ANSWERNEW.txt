//20084
func lexSort(lst) = {
  local arr = copyList(lst)
  for (i: 2 .. arr.listLen()) {
    local key = arr[i]
    local j = i - 1
    while (j >= 1 && arr[j] >> key) {
      arr[j+1] = arr[j]
      j = j - 1
    }
    arr[j+1] = key
  }
  arr
}

func evalRPN(tokens) = {
  local stack = []
  for (t in tokens) {
    if (t === "+") {
      local b = stack[stack.listLen()]; stack.remove(stack.listLen())
      local a = stack[stack.listLen()]; stack.remove(stack.listLen())
      stack.add(a + b)
    } else if (t === "-") {
      local b = stack[stack.listLen()]; stack.remove(stack.listLen())
      local a = stack[stack.listLen()]; stack.remove(stack.listLen())
      stack.add(a - b)
    } else if (t === "*") {
      local b = stack[stack.listLen()]; stack.remove(stack.listLen())
      local a = stack[stack.listLen()]; stack.remove(stack.listLen())
      stack.add(a * b)
    } else if (t === "/") {
      local b = stack[stack.listLen()]; stack.remove(stack.listLen())
      local a = stack[stack.listLen()]; stack.remove(stack.listLen())
      stack.add(a / b)
    } else {
      stack.add(dec(t))
    }
  }
  stack[1]
}

func infixToPostfix(tokens) = {
  local prec = {"+":1,"-":1,"*":2,"/":2}
  local output = []
  local stack = []
  for (t in tokens) {
    if (t === "+" || t === "-" || t === "*" || t === "/") {
      while (!(stack ? emptyList) && stack[stack.listLen()] !== "(" && prec.get(stack[stack.listLen()]) >= prec.get(t)) {
        output.add(stack[stack.listLen()])
        stack.remove(stack.listLen())
      }
      stack.add(t)
    } else if (t === "(") {
      stack.add(t)
    } else if (t === ")") {
      while (stack[stack.listLen()] !== "(") {
        output.add(stack[stack.listLen()])
        stack.remove(stack.listLen())
      }
      stack.remove(stack.listLen())
    } else {
      output.add(t)
    }
  }
  while (!(stack ? emptyList)) {
    output.add(stack[stack.listLen()])
    stack.remove(stack.listLen())
  }
  output
}
println(infixToPostfix(["a","+","b","*","c"]))

      local ch = s.segment(i,1)
      if (ch === cur) cnt = cnt + 1

func tokenizer(s) = {
  local result = []
  local i = 1
  while (i <= s.textLen()) {
    local ch = s.segment(i,1)
    local isDigit = ch ? base10 && ch.textLen() == 1 && !(ch === ".")
    local isLetter = !(ch ? number) && ch !== "+" && ch !== "-" && ch !== "*" && ch !== "/" && ch !== "(" && ch !== ")"
    if (ch ? base10 && !(ch === ".")) {
      local tok = ""
      while (i <= s.textLen() && s.segment(i,1) ? base10) {
        tok = tok _ s.segment(i,1)
        i = i + 1
      }
      result.add({"type":"NUM","val":tok})
    } else if (isLetter) {
      local tok = ""
      while (i <= s.textLen() && !(s.segment(i,1) ? number) && s.segment(i,1) !== "+" && s.segment(i,1) !== "-") {
        tok = tok _ s.segment(i,1)
        i = i + 1
      }
      result.add({"type":"WORD","val":tok})
    } else {
      result.add({"type":"SYM","val":ch})
      i = i + 1
    }
  }
  result
}
println(tokenizer("abc123+def"))

func longestCommonSubsequence(a, b) = {
  local m = a.textLen()
  local n = b.textLen()
  local dp = []
  for (i: 0 .. m) {
    local row = []
    for (j: 0 .. n) row.add(0)
    dp.add(row)

      if (a.segment(i,1) === b.segment(j,1)) dp[i+1][j+1] = dp[i][j] + 1

func knapsack(weights, values, capacity) = {
  local n = weights.listLen()
  local dp = []
  for (i: 0 .. n) {
    local row = []
    for (j: 0 .. capacity) row.add(0)
    dp.add(row)
  }
  for (i: 1 .. n) {
    for (j: 0 .. capacity) {
      dp[i+1][j+1] = dp[i][j+1]
      if (weights[i] <= j) {
        dp[i+1][j+1] = max(dp[i+1][j+1], dp[i][j-weights[i]+1] + values[i])
      }
    }
  }
  dp[n+1][capacity+1]
}

    local bit_b = floor(b / (2^(i-1))) % 2
    local s = bit_a ~ bit_b ~ carry

    local row = []

    top = top + 1
    for (r: top .. bottom) { mat[r][right] = num; num = num + 1 }
    right = right - 1
    for (c: right .. left step -1) { mat[bottom][c] = num; num = num + 1 }
    bottom = bottom - 1
    for (r: bottom .. top step -1) { mat[r][left] = num; num = num + 1 }

  }

}

func dijkstra(graph, start) = {
  local dist = {}
  local visited = {}
  for (node in graph.keys()) dist.set(node, 999999)
  dist.set(start, 0)
  local nodes = graph.keys()
  for (iter: 1 .. nodes.listLen()) {
    local u = ""
    local minD = 999999
    for (node in nodes) {
      if (!visited.containsKey(node) && dist.get(node) < minD) {
        minD = dist.get(node); u = node
      }
    }
    if (u ? emptyText) iter = nodes.listLen()
    else {
      visited.set(u, true)
      for (edge in graph.get(u)) {
        local v = edge[1]; local w = edge[2]
        if (dist.get(u) + w < dist.get(v)) dist.set(v, dist.get(u) + w)
      }
    }
  }
  dist
}

func hasCycle(graph) = {
  local visited = {}
  local recStack = {}
  func dfsCheck(node) = {
    visited.set(node, true)
    recStack.set(node, true)
    local found = false
    for (neighbor in graph.get(node)) {
      if (!visited.containsKey(neighbor)) {
        if (dfsCheck(neighbor)) found = true
      } else if (recStack.containsKey(neighbor)) {
        found = true
      }
    }
    recStack.delete(node)
    found
  }
  local cycle = false
  for (node in graph.keys()) {
    if (!visited.containsKey(node)) {
      if (dfsCheck(node)) cycle = true
    }
  }
  cycle
}

func topoSort(graph) = {
  local inDegree = {}
  for (node in graph.keys()) inDegree.set(node, 0)
  for (node in graph.keys()) {
    for (neighbor in graph.get(node)) {
      inDegree.set(neighbor, inDegree.get(neighbor) + 1)
    }
  }
  local queue = graph.keys().filter { n -> inDegree.get(n) == 0 }
  local order = []
  while (!(queue ? emptyList)) {
    local node = queue[1]; queue.remove(1)
    order.add(node)
    for (neighbor in graph.get(node)) {
      inDegree.set(neighbor, inDegree.get(neighbor) - 1)
      if (inDegree.get(neighbor) == 0) queue.add(neighbor)
    }
  }
  order
}

func kMeans(points, k) = {
  local centroids = points.slice(1, k)
  local assign = points.map { p ->
    local minD = 999999; local minC = 1
    for (i: 1 .. centroids.listLen()) {
      local d = sqrt((p[1]-centroids[i][1])^2 + (p[2]-centroids[i][2])^2)
      if (d < minD) { minD = d; minC = i }
    }
    minC
  }
  local newCentroids = []
  for (i: 1 .. k) {
    local cluster = []
    for (j: 1 .. points.listLen()) { if (assign[j] == i) cluster.add(points[j]) }
    if (!(cluster ? emptyList)) {
      newCentroids.add([cluster.map { p -> p[1] }.reduce(0) { x, s -> x+s } / cluster.listLen(), cluster.map { p -> p[2] }.reduce(0) { x, s -> x+s } / cluster.listLen()])
    } else newCentroids.add(centroids[i])
  }
  {"assignments": assign, "centroids": newCentroids}
}

func linearRegression(x, y) = {

func naiveBayes(trainData, trainLabels, query) = {
  local classes = {}
  for (i: 1 .. trainLabels.listLen()) {
    local label = trainLabels[i]
    if (!classes.containsKey(label)) classes.set(label, {"count":0,"features":{}})
    local c = classes.get(label)
    c.set("count", c.get("count")+1)
    for (j: 1 .. trainData[i].listLen()) {
      local fkey = "" _ j _ "_" _ trainData[i][j]
      c.get("features").set(fkey, if (c.get("features").containsKey(fkey)) c.get("features").get(fkey)+1 else 1)
    }
  }
  local total = trainLabels.listLen()
  local bestLabel = ""
  local bestScore = -999999
  for (label, info in classes) {
    local score = log(info.get("count") / total)
    for (j: 1 .. query.listLen()) {
      local fkey = "" _ j _ "_" _ query[j]
      local cnt = if (info.get("features").containsKey(fkey)) info.get("features").get(fkey) else 0
      score = score + log((cnt+1)/(info.get("count")+2))
    }
    if (score > bestScore) { bestScore = score; bestLabel = label }
  }
  bestLabel
}

func kmpSearch(text, pattern) = {
  local n = text.textLen()
  local m = pattern.textLen()
  local lps = []
  for (i: 1 .. m) lps.add(0)
  local len = 0; local i = 2
  while (i <= m) {
    if (pattern.segment(i,1) === pattern.segment(len+1,1)) {
      len = len + 1; lps[i] = len; i = i + 1
    } else if (len > 0) { len = lps[len] }
    else { lps[i] = 0; i = i + 1 }
  }
  local result = []
  local j = 1; i = 1
  while (i <= n) {
    if (text.segment(i,1) === pattern.segment(j,1)) {
      i = i + 1; j = j + 1
    }
    if (j > m) {
      result.add(i - m)
      j = lps[j-1] + 1
    } else if (i <= n && text.segment(i,1) !== pattern.segment(j,1)) {
      if (j > 1) j = lps[j-1] + 1
      else i = i + 1
    }
  }
  result
}

func heapSort(lst) = {
  local arr = copyList(lst)
  local n = arr.listLen()
  func siftDown(i, end) = {
    local root = i
    while (root*2 <= end) {
      local child = root*2
      if (child+1 <= end && arr[child] < arr[child+1]) child = child+1
      if (arr[root] < arr[child]) {
        local t = arr[root]; arr[root] = arr[child]; arr[child] = t
        root = child
      } else root = end+1
    }
  }
  for (i: floor(n/2) .. 1 step -1) siftDown(i, n)
  for (i: n .. 2 step -1) {
    local t = arr[1]; arr[1] = arr[i]; arr[i] = t
    siftDown(1, i-1)
  }
  arr
}
println(heapSort([4,1,7,3,9,2]))

  }

func damerauLevenshtein(a, b) = {
  local m = a.textLen(); local n = b.textLen()
  local dp = []
  for (i: 0 .. m) {
    local row = []
    for (j: 0 .. n) row.add(0)
    dp.add(row)
  }
  for (i: 0 .. m) dp[i+1][1] = i
  for (j: 0 .. n) dp[1][j+1] = j
  for (i: 1 .. m) {
    for (j: 1 .. n) {
      local cost = if (a.segment(i,1) === b.segment(j,1)) 0 else 1
      dp[i+1][j+1] = min(dp[i][j]+cost, min(dp[i+1][j]+1, dp[i][j+1]+1))
      if (i > 1 && j > 1 && a.segment(i,1) === b.segment(j-1,1) && a.segment(i-1,1) === b.segment(j,1)) {
        dp[i+1][j+1] = min(dp[i+1][j+1], dp[i-1][j-1]+cost)
      }
    }
  }
  dp[m+1][n+1]
}

func minimumWindowSubstring(s, t) = {
  local need = {}
  for (i: 1 .. t.textLen()) {
    local ch = t.segment(i,1)
    need.set(ch, if (need.containsKey(ch)) need.get(ch)+1 else 1)
  }
  local window = {}
  local have = 0; local required = need.dictLen()
  local lo = 1; local result = ""
  local minLen = s.textLen()+1
  for (hi: 1 .. s.textLen()) {
    local ch = s.segment(hi,1)
    window.set(ch, if (window.containsKey(ch)) window.get(ch)+1 else 1)
    if (need.containsKey(ch) && window.get(ch) == need.get(ch)) have = have+1
    while (have == required) {
      if (hi-lo+1 < minLen) { minLen = hi-lo+1; result = s.segment(lo,minLen) }
      local lch = s.segment(lo,1)
      window.set(lch, window.get(lch)-1)
      if (need.containsKey(lch) && window.get(lch) < need.get(lch)) have = have-1
      lo = lo+1
    }
  }
  result
}

func modPow(base, exp, mod) = {
  local result = 1; local b = base % mod; local e = exp
  while (e > 0) {
    if (e % 2 == 1) result = (result*b) % mod
    b = (b*b) % mod
    e = floor(e/2)
  }
  result
}

func millerRabin(n, k) = {
  if (n < 2) false
  else if (n == 2 || n == 3) true
  else if (n % 2 == 0) false
  else {
    local d = n-1; local r = 0
    while (d % 2 == 0) { d = floor(d/2); r = r+1 }
    local prime = true
    for (i: 1 .. k) {
      local a = randInt(2, n-2)
      local x = modPow(a, d, n)
      if (x != 1 && x != n-1) {
        local cont = false
        for (j: 1 .. r-1) {
          x = (x*x) % n
          if (x == n-1) { cont = true; j = r }
        }
        if (!cont) prime = false
      }
    }
    prime
  }
}
println(millerRabin(17,5))
println(millerRabin(15,5))
println(millerRabin(104729,10))

func shortestPath(grid) = {
  local rows = grid.listLen(); local cols = grid[1].listLen()
  if (grid[1][1] == 1 || grid[rows][cols] == 1) -1
  else {
    local dist = {}
    local queue = [[1,1,1]]
    dist.set("1,1", 1)
    while (!(queue ? emptyList)) {
      local curr = queue[1]; queue.remove(1)
      local r = curr[1]; local c = curr[2]; local d = curr[3]
      if (r == rows && c == cols) d
      else {
        for (dir in [[-1,0],[1,0],[0,-1],[0,1]]) {
          local nr = r+dir[1]; local nc = c+dir[2]
          local key = nr _ "," _ nc
          if (nr >= 1 && nr <= rows && nc >= 1 && nc <= cols && grid[nr][nc] == 0 && !dist.containsKey(key)) {
            dist.set(key, d+1)
            queue.add([nr,nc,d+1])
          }
        }
        0
      }
    }
    -1
  }
}

func hamiltonianPath(graph) = {
  local nodes = graph.keys()
  local n = nodes.listLen()
  local found = false
  local path = []
  func bt(current, visited) = {
    if (!found) {
      path.add(current)
      visited.set(current, true)
      if (path.listLen() == n) {
        found = true
      } else {
        for (neighbor in graph.get(current)) {
          if (!visited.containsKey(neighbor) && !found) bt(neighbor, visited)
        }
      }
      if (!found) {
        path.remove(path.listLen())
        visited.delete(current)
      }
    }
  }
  for (node in nodes) {
    if (!found) bt(node, {})
  }
  if (found) path else []
}

func necklaces(n, k) = {
  func gcd496(a, b) = {
    local x = a; local y = b
    while (y != 0) { local t = y; y = x%y; x = t }
    x
  }
  func binomCoeff(n, k) = {
    if (k > n) 0
    else if (k == 0 || k == n) 1
    else {
      local result = 1
      for (i: 1 .. min(k, n-k)) {
        result = result * (n-i+1) / i
      }
      result
    }
  }
  local total = 0
  for (d: 1 .. n) {
    if (n % d == 0) {
      total = total + binomCoeff(d, round(k * d / n))
    }
  }
  round(total / n)
}

func simulatedAnnealing497(fn, init, temp, cooling) = {
  local x = init
  local T = temp
  local best = x
  local bestVal = fn(x)
  for (i: 1 .. 1000) {
    local xNew = x + (randFloat() - 0.5) * T
    local delta = fn(xNew) - fn(x)
    if (delta < 0 || randFloat() < exp(-delta / T)) {
      x = xNew
      if (fn(x) < bestVal) {
        best = x
        bestVal = fn(x)
      }
    }
    T = T * cooling
  }
  {"x_opt": formatDecimal(best, 4), "f_opt": formatDecimal(bestVal, 6)}
}

func geneticStep(pop, fitFn, mutRate) = {
  local scored = pop.map { ind -> [fitFn(ind), ind] }
  local sorted = scored.sort { a, b -> a[1] > b[1] }
  local half = floor(sorted.listLen() / 2)
  local survivors = sorted.slice(1, half).map { s -> s[2] }
  local newPop = copyList(survivors)
  for (i: 1 .. survivors.listLen() - 1 step 2) {
    local p1 = survivors[i]
    local p2 = survivors[i+1]
    local mid = floor(p1.listLen() / 2)
    local child = p1.slice(1, mid)
    child.appendList(p2.slice(mid+1, p2.listLen()))
    local mutated = child.map { bit -> if (randFloat() < mutRate) (bit ~ 1) else bit }
    newPop.add(mutated)
  }
  newPop
}

func antColony(graph, ants, iters) = {
  local nodes = graph.keys()
  local n = nodes.listLen()
  local pheromones = {}
  for (i in nodes) {
    pheromones.set(i, {})
    for (j in nodes) pheromones.get(i).set(j, 1.0)
  }
  local bestPath = []
  local bestLen = 999999
  for (iter: 1 .. iters) {
    for (ant: 1 .. ants) {
      local path = [nodes[randInt(1, n)]]
      local visited = {}
      visited.set(path[1], true)
      while (path.listLen() < n) {
        local cur = path[path.listLen()]
        local neighbors = graph.get(cur)
        local unvisited = neighbors.filter { nb -> !visited.containsKey(nb[1]) }
        if (unvisited ? emptyList) {
          path.listLen()
        } else {
          local totalPher = unvisited.reduce(0) { nb, s -> s + pheromones.get(cur).get(nb[1]) }
          local r = randFloat() * totalPher
          local chosen = unvisited[1]
          local cumul = 0
          for (nb in unvisited) {
            cumul = cumul + pheromones.get(cur).get(nb[1])
            if (cumul >= r) { chosen = nb; cumul = totalPher }
          }
          path.add(chosen[1])
          visited.set(chosen[1], true)
        }
      }
      local pathLen = 0
      for (i: 1 .. path.listLen()-1) {
        local edges = graph.get(path[i])
        for (e in edges) {
          if (e[1] === path[i+1]) pathLen = pathLen + e[2]
        }
      }
      if (pathLen < bestLen && path.listLen() == n) {
        bestLen = pathLen
        bestPath = path
      }
      for (i: 1 .. path.listLen()-1) {
        local cur = path[i]; local nxt = path[i+1]
        pheromones.get(cur).set(nxt, pheromones.get(cur).get(nxt) + 1.0/max(pathLen,1))
      }
    }
    for (i in nodes) {
      for (j in nodes) {
        pheromones.get(i).set(j, pheromones.get(i).get(j) * 0.9)
      }
    }
  }
  {"best_path": bestPath, "best_length": bestLen}
}
