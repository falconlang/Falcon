func sumDigits(n) = {
  local sum = 0
  local temp = n
  while (temp > 0) {
    sum = sum + (temp % 10)
    temp = floor(temp / 10)
  }
  sum
}
func isSmith(n) = {
  local pSum = 0
  local temp = n
  local factors = 0
  for (i: 2 .. floor(sqrt(n))) {
    while (temp % i == 0) {
      pSum = pSum + sumDigits(i)
      temp = temp / i
      factors = factors + 1
    }
  }
  if (temp > 1) {
    pSum = pSum + sumDigits(temp)
    factors = factors + 1
  }
  factors > 1 && sumDigits(n) == pSum
}
for (v in [4, 22, 27, 121]) println(isSmith(v))

func jacobiSymbol(a, n) = {
  local a_val = a % n
  local n_val = n
  local t = 1
  while (a_val != 0) {
    while (a_val % 2 == 0) {
      a_val = a_val / 2
      local r = n_val % 8
      if (r == 3 || r == 5) t = -t
    }
    local temp = a_val
    a_val = n_val
    n_val = temp
    if (a_val % 4 == 3 && n_val % 4 == 3) t = -t
    a_val = a_val % n_val
  }
  if (n_val == 1) t else 0
}
println(jacobiSymbol(2, 15))

  local temp = n
  local chars = "123456789"
  while (temp > 0) {

func smallestWindow(s, t) = {
  // Simplified approximation for standard ASCII window search
  local minLen = 999999
  local bestSub = ""
  for (i: 1 .. s.textLen()) {
    for (j: i .. s.textLen()) {
      local sub = s.segment(i, j - i + 1)
      local valid = true
      for (k: 1 .. t.textLen()) {
        if (!sub.contains(t.segment(k, 1))) {
          valid = false
          break
        }
      }
      if (valid && sub.textLen() < minLen) {
        minLen = sub.textLen()
        bestSub = sub
      }
    }
  }
  bestSub
}
println(smallestWindow("ADOBECODEBANC", "ABC"))

func toCurrencyString(amount) = {
  local parts = (amount _ "").split(".")
  local intPart = parts[1]
  local res = ""
  local c = 0
  for (i: intPart.textLen() .. 1 step -1) {
    if (c > 0 && c % 3 == 0) res = "," _ res
    res = intPart.segment(i, 1) _ res
    c = c + 1
  }
  if (parts.listLen() > 1) res _ "." _ parts[2] else res
}

func slidingWindowMax(list, k) = {
  local res =[]
  if (list.listLen() == 0 || k == 0) res else {
    for (i: 1 .. list.listLen() - k + 1) {
      local mx = list[i]
      for (j: 1 .. k - 1) {
        if (list[i + j] > mx) mx = list[i + j]
      }
      res.add(mx)
    }
    res
  }
}

func mergeWithConflict(a, b, strategy) = {
  local res = copyDict(a)
  for (k, v in b) {
    if (res.containsKey(k)) {
      if (strategy === "first") res.set(k, a.get(k))
      else if (strategy === "second") res.set(k, v)
      else if (strategy === "sum") res.set(k, a.get(k) + v)
    } else {
      res.set(k, v)
    }
  }
  res
}

local d2_1117 = {"y": 3, "z": 4}

  local res = headers.join(",") _ "\n"
  for (row in rows) {
    local r =[]

  }

}

func heapPush(heap, val) = {
  heap.add(val)
  local i = heap.listLen()
  while (i > 1 && heap[floor(i / 2)] > heap[i]) {
    local temp = heap[floor(i / 2)]
    heap[floor(i / 2)] = heap[i]
    heap[i] = temp
    i = floor(i / 2)
  }
}
func heapPop(heap) = {
  local mn = heap[1]
  heap[1] = heap[heap.listLen()]
  heap.remove(heap.listLen())
  local i = 1
  while (i * 2 <= heap.listLen()) {
    local mc = i * 2
    if (mc + 1 <= heap.listLen() && heap[mc + 1] < heap[mc]) mc = mc + 1
    if (heap[i] <= heap[mc]) break
    local temp = heap[i]
    heap[i] = heap[mc]
    heap[mc] = temp
    i = mc
  }
  mn
}
local myHeap =[]
heapPush(myHeap, 3)
heapPush(myHeap, 1)
heapPush(myHeap, 2)
println(heapPop(myHeap))

func floydWarshall(matrix) = {
  local dist = copyList(matrix)
  local n = matrix.listLen()
  for (k: 1 .. n) {
    for (i: 1 .. n) {
      for (j: 1 .. n) {
        if (dist[i][k] + dist[k][j] < dist[i][j]) {
          dist[i][j] = dist[i][k] + dist[k][j]
        }
      }
    }
  }
  dist
}
local inf = 999999
println(floydWarshall([[0, 5, inf, 10],
  [inf, 0, 3, inf],[inf, inf, 0, 1],
  [inf, inf, inf, 0]
]))

func geneticStep(population, target, mutRate) = {
  local nextGen =[]
  for (i: 1 .. population.listLen() / 2) {
    local p1 = population[randInt(1, population.listLen())]
    local p2 = population[randInt(1, population.listLen())]
    local child = ""
    for (j: 1 .. p1.textLen()) {
      local bit = if (randFloat() < 0.5) p1.segment(j, 1) else p2.segment(j, 1)
      if (randFloat() < mutRate) bit = if (bit === "0") "1" else "0"
      child = child _ bit
    }
    nextGen.add(child)
    nextGen.add(child)
  }
  nextGen
}
println(geneticStep(["1010", "1100", "0011", "0101"], "1111", 0.1))
