func modInv(a, m) = {
  local old_r = a
  local r = m
  local old_s = 1
  local s = 0
  while (r != 0) {
    local q = floor(old_r / r)
    local tmp_r = r
    r = old_r - q * r
    old_r = tmp_r
    local tmp_s = s
    s = old_s - q * s
    old_s = tmp_s
  }
  mod(old_s + m, m)
}

func millerRabin(n, witnesses) = {
  if (n < 2) { false }
  else if (n == 2) { true }
  else if (mod(n,2)==0) { false }
  else {
    local d = n-1
    local r = 0
    while (mod(d,2)==0) { d=floor(d/2) r=r+1 }
    local isPrime = true
    for (a in witnesses) {
      if (a >= n) {  }
      else {
        local x = modPow(a, d, n)
        if (x != 1 && x != n-1) {
          local cont = true
          for (j: 1 .. r-1) {
            if (cont) {
              x = modPow(x, 2, n)
              if (x == n-1) { cont = false }
            }
          }
          if (cont) { isPrime = false }
        }
      }
    }
    isPrime
  }
}

func tonelliShanks(n, p) = {
  if (modPow(n, floor((p-1)/2), p) != 1) { -1 }
  else {
    local q = p - 1
    local s = 0
    while (mod(q,2)==0) { q=floor(q/2) s=s+1 }
    if (s == 1) { modPow(n, floor((p+1)/4), p) }
    else {
      local z = 2
      while (modPow(z, floor((p-1)/2), p) != p-1) { z = z+1 }
      local m2 = s
      local c2 = modPow(z, q, p)
      local t2 = modPow(n, q, p)
      local r2 = modPow(n, floor((q+1)/2), p)
      local result = -1
      local cont = true
      while (cont) {
        if (t2 == 0) { result = 0 cont = false }
        else if (t2 == 1) { result = r2 cont = false }
        else {
          local i2 = 1
          local tmp = mod(t2*t2, p)
          while (tmp != 1) { tmp = mod(tmp*tmp,p) i2=i2+1 }
          local b2 = modPow(c2, 2^(m2-i2-1), p)
          m2 = i2
          c2 = mod(b2*b2,p)
          t2 = mod(t2*c2,p)
          r2 = mod(r2*b2,p)
        }
      }
      result
    }
  }
}

  local ipad = key.map { b -> b ~ 0x36 }
  local opad = key.map { b -> b ~ 0x5c }
  local inner = 0
  for (b in ipad) { inner = mod(inner * 31 + b, 4294967296) }
  for (b in message) { inner = mod(inner * 31 + b, 4294967296) }
  local outer = 0
  for (b in opad) { outer = mod(outer * 31 + b, 4294967296) }
  outer = mod(outer * 31 + inner, 4294967296)
  outer
}
println(hmacSHA256_simple([0x0b,0x0b,0x0b,0x0b],[0x48,0x69,0x20,0x54,0x68,0x65,0x72,0x65]))

// 10160
func zkProof_sim(statement, witness, verifier) = {
  local commitment = mod(witness * witness, statement)
  local challenge = randInt(0, statement-1)
  local response = mod(witness + challenge, statement)
  local check = mod(response * response, statement) == mod(commitment + challenge * statement, statement)
  if (check || mod(witness*witness,statement)==mod(commitment,1)) "verified" else "rejected"
}
println(zkProof_sim(97, 5, 1))

// 10166
func penetrationTest_sim(target, techniques) = {
  local findings = {}
  for (tech in techniques) {
    local vuln = randFloat() < 0.3
    findings.set(tech, {"target":target,"technique":tech,"vulnerable":vuln,"severity":if (vuln) "high" else "none"})
  }
  findings
}
println(penetrationTest_sim("192.168.1.1",["sql_injection","xss","csrf","path_traversal"]))

func complianceAudit(controls_dict, framework) = {
  local coverage = 0
  local total = controls_dict.dictLen()
  for (id, ctrl in controls_dict) {
    if (ctrl.get("status","") === "pass") { coverage = coverage+1 }
  }
  {"framework":framework,"total":total,"passed":coverage,"coverage":if (total>0) formatDecimal(coverage/total*100,1) else "0.0"}
}
local controls10172 = {"AC-1":{"status":"pass"},"AC-2":{"status":"fail"},"AC-3":{"status":"pass"}}
println(complianceAudit(controls10172,"NIST"))

func secureChannel_establish(alice, bob) = {
  local g = 2
  local p = 23
  local aSecret = randInt(2, p-2)
  local bSecret = randInt(2, p-2)
  local aPub = modPow(g, aSecret, p)
  local bPub = modPow(g, bSecret, p)
  local sharedKey = modPow(bPub, aSecret, p)
  {"alice":alice,"bob":bob,"sharedKey":sharedKey,"established":true}
}
println(secureChannel_establish("alice","bob"))

// 10185
func cryptoProtocol_sim(protocol_name, parties, messages) = {
  local steps = []
  if (protocol_name === "TLS") {
    steps.add("ClientHello")
    steps.add("ServerHello")
    steps.add("Certificate")
    steps.add("KeyExchange")
    steps.add("Finished")
  } else if (protocol_name === "SSH") {
    steps.add("VersionExchange")
    steps.add("KeyInit")
    steps.add("Auth")
    steps.add("Session")
  }
  {"protocol":protocol_name,"parties":parties,"steps":steps,"messages":messages}
}
println(cryptoProtocol_sim("TLS",["client","server"],[]))

func attestation_sim(device, nonce, policy) = {
  local measurements = {"firmware":device.get("firmwareHash",""),"boot":device.get("bootChain","secure")}

  local policyMet = policy.get("requireSecureBoot",false) && valid

}

func cryptoBenchmark(algorithms, key_sizes, data_size) = {
  local results = {}
  for (algo in algorithms) {
    local algoResults = {}
    for (ks in key_sizes) {
      local baseOps = if (algo === "AES") 1000 else if (algo === "RSA") 10 else 500
      local throughput = baseOps * data_size / ks
      algoResults.set(ks, formatDecimal(throughput,2))
    }
    results.set(algo, algoResults)
  }
  results
}
println(cryptoBenchmark(["AES","RSA","ChaCha20"],[128,256,512],1024))
