func tspGreedy(dist_matrix, n) = {
  local visited = [1]
  local tour = [1]
  while (visited.listLen() < n) {
    local last = tour[tour.listLen()]
    local best = 999999
    local bestNode = -1
    for (j: 1 .. n) {
      if (!visited.containsItem(j) && dist_matrix[last][j] < best) {
        best = dist_matrix[last][j]
        bestNode = j
      }
    }
    visited.add(bestNode)
    tour.add(bestNode)
  }
  tour.add(1)
  tour
}

    x
  } else {
    local root = ufFind(this.ufParent.get(x, x))

  }
}

      if (!dist.containsKey(nb)) {
        dist.set(nb, dist.get(cur, 0) + 1)

func networkReliability(graph, p) = {
  local nodes = graph.keys()
  local n = nodes.listLen()
  local edges = []
  for (u, neighbors in graph) {
    for (v in neighbors) {
      if (u << v) {
        edges.add([u, v])
      }
    }
  }
  local connected = 0
  for (trial: 1 .. 100) {
    local activeEdges = {}
    for (nd in nodes) { activeEdges.set(nd, []) }
    for (e in edges) {
      if (randFloat() < p) {
        activeEdges.get(e[1], []).add(e[2])
        activeEdges.get(e[2], []).add(e[1])
      }
    }
    local visited = []
    local queue = [nodes[1]]
    visited.add(nodes[1])
    while (queue.listLen() > 0) {
      local cur = queue[1]
      queue.remove(1)
      for (nb in activeEdges.get(cur, [])) {
        if (!visited.containsItem(nb)) {
          visited.add(nb)
          queue.add(nb)
        }
      }
    }
    if (visited.listLen() == n) {
      connected = connected + 1
    }
  }
  connected / 100
}

func dijkstra(graph, start) = {
  local dist = {}
  local prev = {}
  local visited = []
  for (n, edges in graph) {
    dist.set(n, 999999)
  }
  dist.set(start, 0)
  local queue = [start]
  while (queue.listLen() > 0) {
    local best = 999999
    local cur = queue[1]
    for (n in queue) {
      if (dist.get(n, 999999) < best) {
        best = dist.get(n, 999999)
        cur = n
      }
    }
    queue.remove(queue.indexOf(cur))
    visited.add(cur)
    for (nb, w in graph.get(cur, {})) {
      local newDist = dist.get(cur, 999999) + w
      if (newDist < dist.get(nb, 999999)) {
        dist.set(nb, newDist)
        prev.set(nb, cur)
        if (!visited.containsItem(nb) && !queue.containsItem(nb)) {
          queue.add(nb)
        }
      }
    }
  }
  [dist, prev]
}
func shortestPathStr(graph, u, v) = {
  local dp = dijkstra(graph, u)
  local dist = dp[1]
  local prev = dp[2]
  local path = []
  local cur = v
  while (cur !== u) {
    path.add(cur)
    cur = prev.get(cur, u)
  }
  path.add(u)
  path.reverseList()
  path.join("→") _ " (cost: " _ dist.get(v, 999999) _ ")"
}

func treeProfile(tree, root) = {
  local height = 0
  local leafCount = 0
  local internalCount = 0
  local branchSum = 0
  func dfs(node, depth) {
    local children = tree.get(node, [])

      for (child in children) {

    }
  }
  dfs(root, 0)
  local bf = if (internalCount > 0) branchSum / internalCount else 0
  "Height: " _ height _ "\nLeaves: " _ leafCount _ "\nInternal: " _ internalCount _ "\nBranching factor: " _ bf

func independentSets(graph) = {
  local nodes = graph.keys()
  local n = nodes.listLen()
  local results = []
  func isIndependent(subset) = {
    local ok = true
    for (i: 1 .. subset.listLen()) {
      for (j: i+1 .. subset.listLen()) {
        if (graph.get(subset[i], []).containsItem(subset[j])) {
          ok = false
        }
      }
    }
    ok
  }
  func isMaximal(subset) = {
    local maximal = true
    for (nd in nodes) {
      if (!subset.containsItem(nd)) {
        local canAdd = true
        for (s in subset) {
          if (graph.get(nd, []).containsItem(s)) {
            canAdd = false
          }
        }
        if (canAdd) { maximal = false }
      }
    }
    maximal
  }
  for (mask: 0 .. (1 << n) - 1) {
    local subset = []
    for (i: 1 .. n) {
      if (floor(mask / (2 ^ (i-1))) % 2 == 1) {
        subset.add(nodes[i])
      }
    }
    if (isIndependent(subset) && isMaximal(subset)) {
      results.add(subset)
    }
  }
  results
}

func graphIsomorphisms(g1, g2) = {
  local n1 = g1.keys()
  local n2 = g2.keys()
  if (n1.listLen() != n2.listLen()) {
    []
  } else {
    local n = n1.listLen()
    local results = []
    func permute(perm, remaining) {
      if (remaining.listLen() == 0) {
        local valid = true
        for (i: 1 .. n) {
          for (j: 1 .. n) {
            local u = n1[i]
            local v = n1[j]
            local pu = perm.get(u, "")
            local pv = perm.get(v, "")
            local e1 = g1.get(u, []).containsItem(v)
            local e2 = g2.get(pu, []).containsItem(pv)
            if (e1 != e2) { valid = false }
          }
        }
        if (valid) { results.add(copyDict(perm)) }
      } else {
        local src = n1[n - remaining.listLen() + 1]
        for (i: 1 .. remaining.listLen()) {
          perm.set(src, remaining[i])
          local rem2 = copyList(remaining)
          rem2.remove(i)
          permute(perm, rem2)
          perm.delete(src)
        }
      }
    }
    permute({}, copyList(n2))
    results
  }
}

    if (!color.containsKey(start) && isBipartite) {
      local queue = [start]

      while (queue.listLen() > 0 && isBipartite) {
        local cur = queue[1]

            queue.add(nb)
          } else if (color.get(nb, 0) == color.get(cur, 0)) {
            isBipartite = false
          }
        }
      }
    }
  }
  if (isBipartite) { color } else { "NOT_BIPARTITE" }
}

func kruskal(edges, n) = {
  local parent = {}
  for (i: 1 .. n) { parent.set(i, i) }
  func find(x) = {
    if (parent.get(x, x) == x) { x } else {
      local r = find(parent.get(x, x))
      parent.set(x, r)
      r
    }
  }
  func unite(x, y) {

func christofidesApprox(dist, n) = {
  local inMST = [1]
  local inSet = []
  for (i: 1 .. n) { inSet.add(false) }
  inSet[1] = true
  local parent = []
  for (i: 1 .. n) { parent.add(1) }
  local key = []
  for (i: 1 .. n) { key.add(999999) }
  key[1] = 0
  for (iter: 1 .. n - 1) {
    local minKey = 999999
    local u = -1
    for (i: 1 .. n) {
      if (!inSet[i] && key[i] < minKey) {
        minKey = key[i]
        u = i
      }
    }
    inSet[u] = true
    for (v: 1 .. n) {
      if (!inSet[v] && dist[u][v] < key[v]) {
        key[v] = dist[u][v]
        parent[v] = u
      }
    }
  }
  local mstAdj = {}
  for (i: 1 .. n) { mstAdj.set(i, []) }
  for (i: 2 .. n) {
    mstAdj.get(i, []).add(parent[i])
    mstAdj.get(parent[i], []).add(i)
  }
  local oddNodes = []
  for (i: 1 .. n) {
    if (mstAdj.get(i, []).listLen() % 2 != 0) {
      oddNodes.add(i)
    }
  }
  local matched = []
  for (i: 1 .. oddNodes.listLen() step 2) {
    if (i + 1 <= oddNodes.listLen()) {
      mstAdj.get(oddNodes[i], []).add(oddNodes[i+1])
      mstAdj.get(oddNodes[i+1], []).add(oddNodes[i])
      matched.add([oddNodes[i], oddNodes[i+1]])
    }
  }
  local visited = []
  local tour = []
  local stack = [1]
  while (stack.listLen() > 0) {
    local cur = stack[stack.listLen()]
    local nb = mstAdj.get(cur, [])
    if (nb.listLen() == 0) {
      stack.remove(stack.listLen())
      if (!tour.containsItem(cur)) {
        tour.add(cur)
      }
    } else {
      local next = nb[1]
      mstAdj.get(cur, []).remove(1)
      local idx = mstAdj.get(next, []).indexOf(cur)
      if (idx > 0) { mstAdj.get(next, []).remove(idx) }
      stack.add(next)
    }
  }
  tour
}

func graphFeedback(graph) = {

  local remaining = copyDict(graph)
  local changed = true
  while (changed) {
    changed = false
    local nodes = remaining.keys()
    for (n in nodes) {
      local inDeg = 0

        if (neighbors.containsItem(n)) {
          inDeg = inDeg + 1
        }
      }
      if (inDeg == 0 || remaining.get(n, []).listLen() == 0) {
        remaining.delete(n)
        changed = true
      }
    }
  }
  for (u, neighbors in remaining) {
    for (v in neighbors) {
      removed.add([u, v])
    }
  }
  removed
}

func parallelBFS(graph, sources) = {

  local queue = []
  for (s in sources) {
    dist.set(s, 0)
    queue.add(s)
  }
  while (queue.listLen() > 0) {

    queue.remove(1)
    for (nb in graph.get(cur, [])) {
      if (!dist.containsKey(nb)) {
        dist.set(nb, dist.get(cur, 0) + 1)

    }
  }
  dist
}

func dominatorTree(graph, source) = {
  local nodes = graph.keys()
  local dominators = {}
  for (n in nodes) {
    if (n === source) {
      dominators.set(n, source)
    } else {
      local tempGraph = copyDict(graph)
      tempGraph.delete(n)
      local reachable = []
      local queue = [source]
      while (queue.listLen() > 0) {
        local cur = queue[1]
        queue.remove(1)
        if (!reachable.containsItem(cur)) {
          reachable.add(cur)
          for (nb in tempGraph.get(cur, [])) {
            if (!reachable.containsItem(nb)) {
              queue.add(nb)
            }
          }
        }
      }
      if (!reachable.containsItem(n)) {
        dominators.set(n, source)
      } else {
        dominators.set(n, "none")
      }
    }
  }
  dominators
}

func tsp2Opt(tour, dist_matrix) = {
  local improved = copyList(tour)
  local n = tour.listLen() - 1
  for (i: 1 .. n - 1) {
    for (j: i+2 .. n) {
      local d1 = dist_matrix[improved[i]][improved[i+1]] + dist_matrix[improved[j]][improved[j+1]]
      local d2 = dist_matrix[improved[i]][improved[j]] + dist_matrix[improved[i+1]][improved[j+1]]
      if (d2 < d1) {
        local seg = []
        for (k: i+1 .. j) { seg.add(improved[k]) }
        seg.reverseList()
        for (k: i+1 .. j) { improved[k] = seg[k-i] }
      }
    }
  }
  improved
}

func flowDict(network, source, sink) = {
  local flow = {}
  for (u, edges in network) {
    flow.set(u, {})
    for (v, cap in edges) {
      flow.get(u, {}).set(v, 0)
    }
  }
  local totalFlow = 0
  local running = true
  while (running) {
    local parent = {}
    local visited = [source]
    local queue = [source]
    while (queue.listLen() > 0) {
      local cur = queue[1]
      queue.remove(1)
      for (v, cap in network.get(cur, {})) {
        local f = flow.get(cur, {}).get(v, 0)
        if (!visited.containsItem(v) && cap - f > 0) {
          parent.set(v, cur)
          visited.add(v)
          queue.add(v)
        }
      }
    }
    if (!visited.containsItem(sink)) {
      running = false
    } else {
      local pathFlow = 999999
      local cur = sink
      while (cur !== source) {
        local prev = parent.get(cur, "")
        local cap = network.get(prev, {}).get(cur, 0)
        local f = flow.get(prev, {}).get(cur, 0)
        if (cap - f < pathFlow) { pathFlow = cap - f }
        cur = prev
      }
      cur = sink
      while (cur !== source) {
        local prev = parent.get(cur, "")
        flow.get(prev, {}).set(cur, flow.get(prev, {}).get(cur, 0) + pathFlow)
        flow.get(cur, {}).set(prev, flow.get(cur, {}).get(prev, 0) - pathFlow)
        cur = prev
      }
      totalFlow = totalFlow + pathFlow
    }
  }
  flow
}
local net = {"S":{"A":10,"B":10},"A":{"T":10},"B":{"T":10},"T":{}}
println(flowDict(net, "S", "T"))

func clusteringCoeff(graph, node) = {
  local neighbors = graph.get(node, [])
  local k = neighbors.listLen()
  if (k < 2) {
    0
  } else {
    local links = 0
    for (i: 1 .. k) {
      for (j: i+1 .. k) {
        if (graph.get(neighbors[i], []).containsItem(neighbors[j])) {
          links = links + 1
        }
      }
    }
    2 * links / (k * (k - 1))
  }
}
func avgClusteringCoeff(graph) = {
  local nodes = graph.keys()
  local total = 0
  for (n in nodes) {
    total = total + clusteringCoeff(graph, n)
  }
  total / nodes.listLen()
}
local g = {"A":["B","C","D"],"B":["A","C"],"C":["A","B","D"],"D":["A","C"]}
println(avgClusteringCoeff(g))
