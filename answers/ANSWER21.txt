func rosenbrock(x) = {
  (1 - x[1]) ^ 2 + 100 * (x[2] - x[1] ^ 2) ^ 2

func rosenGrad(x) = {
  local g1 = -2 * (1 - x[1]) - 400 * x[1] * (x[2] - x[1]^2)
  local g2 = 200 * (x[2] - x[1]^2)
  [g1, g2]
}

local H_inv = [[1, 0], [0, 1]]
local iter_b = 0

while (iter_b < iters_bfgs) {
  local g = rosenGrad(x_bfgs)
  local p = [-(H_inv[1][1]*g[1] + H_inv[1][2]*g[2]),
             -(H_inv[2][1]*g[1] + H_inv[2][2]*g[2])]
  local alpha = lr_bfgs
  local x_new = [x_bfgs[1] + alpha*p[1], x_bfgs[2] + alpha*p[2]]
  local g_new = rosenGrad(x_new)
  local s = [x_new[1]-x_bfgs[1], x_new[2]-x_bfgs[2]]
  local y = [g_new[1]-g[1], g_new[2]-g[2]]
  local sy = s[1]*y[1] + s[2]*y[2]
  if (abs(sy) > 1e-10) {
    local rho = 1 / sy

func matMul(A, B) = {
  local n = A.listLen()
  local m = B[1].listLen()
  local p = B.listLen()
  local C = []
  for (i: 1 .. n) {
    local row = []
    for (j: 1 .. m) {
      local s = 0
      for (k: 1 .. p) {
        s = s + A[i][k] * B[k][j]
      }
      row.add(s)
    }
    C.add(row)
  }
  C
}

func matPow(A, n) = {
  local result = [[1,0],[0,1]]
  local base = A
  local exp_n = n
  while (exp_n > 0) {
    if (exp_n % 2 == 1) {
      result = matMul(result, base)
    }
    base = matMul(base, base)
    exp_n = floor(exp_n / 2)
  }
  result
}

local M = [[1,1],[1,0]]
println(matPow(M, 5))

func squaredExpKernel(x1, x2) = {
  exp(-0.5 * (x1 - x2) ^ 2)
}

func predict(gp, x_new) = {
  local xs = gp.get("xs")
  local ys = gp.get("ys")
  local n = xs.listLen()
  local k_star = xs.map { xi -> squaredExpKernel(x_new, xi) }
  local K = []
  for (i: 1 .. n) {
    local row = []
    for (j: 1 .. n) {
      row.add(squaredExpKernel(xs[i], xs[j]) + if (i == j) 1e-6 else 0)
    }
    K.add(row)
  }
  local w = k_star
  local mean_pred = 0
  for (i: 1 .. n) {
    mean_pred = mean_pred + w[i] * ys[i]
  }
  mean_pred
}

func secant(f_name, x0, x1, tol, maxIter) = {
  local a = x0
  local b = x1
  local iter_s = 0
  local root = b
  while (iter_s < maxIter) {
    local fa = a^3 - a - 2
    local fb = b^3 - b - 2
    if (abs(fb - fa) < 1e-15) {
      iter_s = maxIter
    }
    root = b - fb * (b - a) / (fb - fa)
    a = b
    b = root
    iter_s = iter_s + 1
    if (abs(b - a) < tol) {
      iter_s = maxIter
    }
  }
  root
}

println("Secant root of x^3-x-2: " _ secant("cubic", 1.0, 2.0, 1e-8, 50))

func muller(f_name, x0, x1, x2, maxIter) = {
  local p0 = x0
  local p1 = x1
  local p2 = x2
  local root_m = p2
  local iter_m = 0
  while (iter_m < maxIter) {
    local f0 = p0^3 - p0 - 2
    local f1 = p1^3 - p1 - 2
    local f2 = p2^3 - p2 - 2
    local h1 = p1 - p0
    local h2 = p2 - p1
    local d1 = (f1 - f0) / h1
    local d2 = (f2 - f1) / h2
    local a_m = (d2 - d1) / (h2 + h1)
    local b_m = a_m * h2 + d2
    local c_m = f2
    local disc = b_m^2 - 4*a_m*c_m
    local denom = if (abs(b_m + sqrt(abs(disc))) > abs(b_m - sqrt(abs(disc)))) b_m + sqrt(abs(disc)) else b_m - sqrt(abs(disc))
    if (abs(denom) < 1e-15) {
      iter_m = maxIter
    }
    local dx = -2 * c_m / denom
    root_m = p2 + dx
    p0 = p1
    p1 = p2
    p2 = root_m
    iter_m = iter_m + 1
    if (abs(dx) < 1e-8) {
      iter_m = maxIter
    }
  }
  root_m
}

println("Muller root: " _ muller("cubic", 0, 1, 2, 50))

func gaussianElim(A, b) = {
  local n = b.listLen()
  local Aug = []
  for (i: 1 .. n) {
    local row = copyList(A[i])
    row.add(b[i])
    Aug.add(row)
  }
  for (col: 1 .. n) {
    local maxRow = col
    local maxVal = abs(Aug[col][col])
    for (row: col+1 .. n) {
      if (abs(Aug[row][col]) > maxVal) {
        maxVal = abs(Aug[row][col])
        maxRow = row
      }
    }
    local tmp = Aug[col]
    Aug[col] = Aug[maxRow]
    Aug[maxRow] = tmp
    for (row: col+1 .. n) {
      local factor = Aug[row][col] / Aug[col][col]
      for (j: col .. n+1) {
        Aug[row][j] = Aug[row][j] - factor * Aug[col][j]
      }
    }
  }
  local x_sol = []
  for (i: 1 .. n) {
    x_sol.add(0)
  }
  for (i: n .. 1 step -1) {
    local s = Aug[i][n+1]
    for (j: i+1 .. n) {
      s = s - Aug[i][j] * x_sol[j]
    }
    x_sol[i] = s / Aug[i][i]
  }
  x_sol
}

println(gaussianElim([[2,1,-1],[3,3,9],[3,3,5]], [8,28,10]))

func qrDecompose(A) = {
  local n = A.listLen()
  local m = A[1].listLen()
  local Q = []
  local R_qr = []
  for (i: 1 .. n) {
    local row = []
    for (j: 1 .. m) {
      row.add(0)
    }
    Q.add(row)
    R_qr.add(copyList(row))
  }
  local cols = []
  for (j: 1 .. m) {
    local col = []
    for (i: 1 .. n) {
      col.add(A[i][j])
    }
    cols.add(col)
  }
  local q_cols = []
  for (j: 1 .. m) {
    local v = copyList(cols[j])
    for (k: 1 .. j-1) {
      local dot = 0
      for (i: 1 .. n) {
        dot = dot + q_cols[k][i] * cols[j][i]
      }
      R_qr[k][j] = dot
      for (i: 1 .. n) {
        v[i] = v[i] - dot * q_cols[k][i]
      }
    }
    local norm_v = sqrt(v.reduce(0) { x, s -> x*x + s })
    R_qr[j][j] = norm_v
    local qj = v.map { vi -> vi / norm_v }
    q_cols.add(qj)
  }
  for (j: 1 .. m) {
    for (i: 1 .. n) {
      Q[i][j] = q_cols[j][i]
    }
  }
  {"Q": Q, "R": R_qr}
}

println(qrDecompose([[1,2],[3,4],[5,6]]))

  local n = xs.listLen()
  local p = degree + 1
  local A_ls = []
  for (i: 1 .. n) {
    local row = []
    for (j: 0 .. degree) {
      row.add(xs[i] ^ j)
    }
    A_ls.add(row)

  local ATA = []
  for (i: 1 .. p) {
    local row = []
    for (j: 1 .. p) {

      row.add(s)
    }
    ATA.add(row)
  }
  local ATb = []
  for (i: 1 .. p) {
    local s = 0
    for (k: 1 .. n) {
      s = s + A_ls[k][i] * ys[k]
    }
    ATb.add(s)

func powerSpectrum(signal) = {
  local n = signal.listLen()
  local ps = []
  for (k: 1 .. n) {
    local re = 0
    local im = 0
    for (t: 1 .. n) {
      local angle = 2 * 3.14159265 * (k-1) * (t-1) / n
      re = re + signal[t] * cos(angle)
      im = im - signal[t] * sin(angle)
    }
    ps.add((re*re + im*im) / n)
  }
  ps
}

  local pbest_val = []
  for (i: 1 .. n_particles) {
    local p = lo + randFloat() * (hi - lo)
    pos.add(p)
    vel.add((randFloat() - 0.5))
    pbest.add(p)
    pbest_val.add(p^2)
  }
  local gbest = pos[1]
  local gbest_val = pos[1]^2
  for (i: 1 .. n_particles) {
    if (pbest_val[i] < gbest_val) {
      gbest = pbest[i]
      gbest_val = pbest_val[i]
    }
  }
  for (iter_p: 1 .. iters) {
    for (i: 1 .. n_particles) {
      vel[i] = 0.7*vel[i] + 1.5*randFloat()*(pbest[i]-pos[i]) + 1.5*randFloat()*(gbest-pos[i])
      pos[i] = pos[i] + vel[i]
      local val = pos[i]^2
      if (val < pbest_val[i]) {
        pbest[i] = pos[i]
        pbest_val[i] = val
        if (val < gbest_val) {
          gbest = pos[i]
          gbest_val = val
        }
      }
    }
  }
  {"x_opt": gbest, "f_opt": gbest_val}
}

func fourier_series_coeffs(f_name, n, period) = {
  local T = period
  local coeffs = []
  local N_quad = 1000
  local dt = T / N_quad
  for (k: 0 .. n) {
    local a_k = 0
    local b_k = 0
    for (j: 0 .. N_quad-1) {
      local t = j * dt
      local ft = if (t < T/2) 1 else -1
      a_k = a_k + ft * cos(2 * 3.14159265 * k * t / T) * dt
      b_k = b_k + ft * sin(2 * 3.14159265 * k * t / T) * dt
    }
    coeffs.add({"k": k, "a": 2*a_k/T, "b": 2*b_k/T})
  }
  coeffs
}

func finite_element_1d(N, bc_left, bc_right) = {

  F_fem[1] = bc_left * 1e30

func thomas_algorithm(a, b, c, d) = {
  local n = b.listLen()
  local c_prime = copyList(c)
  local d_prime = copyList(d)
  c_prime[1] = c[1] / b[1]
  d_prime[1] = d[1] / b[1]
  for (i: 2 .. n) {
    local denom = b[i] - a[i] * c_prime[i-1]
    c_prime[i] = if (i < n) c[i] / denom else 0
    d_prime[i] = (d[i] - a[i] * d_prime[i-1]) / denom
  }
  local x_th = []
  for (i: 1 .. n) {
    x_th.add(0)
  }
  x_th[n] = d_prime[n]
  for (i: n-1 .. 1 step -1) {
    x_th[i] = d_prime[i] - c_prime[i] * x_th[i+1]
  }
  x_th
}

func propagateError(formula_name, values, errors) = {
  local result = 0
  if (formula_name === "sum") {
    local err_sq = errors.reduce(0) { e, s -> e*e + s }
    result = sqrt(err_sq)
  } else if (formula_name === "product") {
    local rel_sq = 0
    for (i: 1 .. values.listLen()) {
      rel_sq = rel_sq + (errors[i] / values[i])^2
    }
    local prod = values.reduce(1) { v, s -> v * s }
    result = prod * sqrt(rel_sq)
  }
  {"formula": formula_name, "propagated_error": result}
}

println(propagateError("product", [10, 20], [0.5, 0.3]))

func kaplan_meier(times, events) = {
  local n = times.listLen()
  local sorted_t = copyList(times)
  sorted_t.sort()
  local unique_t = []
  for (t in sorted_t) {
    if (unique_t.listLen() == 0 || unique_t[unique_t.listLen()] != t) {
      unique_t.add(t)
    }
  }
  local S = 1.0
  local curve_t = []
  local curve_s = []
  local n_at_risk = n
  for (tj in unique_t) {
    local d_j = 0
    local c_j = 0
    for (i: 1 .. n) {
      if (times[i] == tj) {
        if (events[i] == 1) {
          d_j = d_j + 1
        } else {
          c_j = c_j + 1
        }
      }
    }
    S = S * (1 - d_j / n_at_risk)
    n_at_risk = n_at_risk - d_j - c_j
    curve_t.add(tj)
    curve_s.add(S)
  }
  {"t": curve_t, "S": curve_s}
}

func assemble(fem) = {
  local N_nodes = fem.get("mesh").listLen()
  local K_f = []
  local F_f = []
  local h = 1.0 / (N_nodes - 1)
  for (i: 1 .. N_nodes) {
    local row = []
    for (j: 1 .. N_nodes) {
      row.add(0)
    }
    K_f.add(row)
    F_f.add(h)
  }
  for (e: 1 .. N_nodes-1) {
    K_f[e][e] = K_f[e][e] + 1/h
    K_f[e][e+1] = K_f[e][e+1] - 1/h
    K_f[e+1][e] = K_f[e+1][e] - 1/h
    K_f[e+1][e+1] = K_f[e+1][e+1] + 1/h
  }
  K_f[1][1] = 1e30
  F_f[1] = 0
  K_f[N_nodes][N_nodes] = 1e30
  F_f[N_nodes] = 0
  fem.set("K", K_f)
  fem.set("F", F_f)
  fem
}

  gaussianElim(fem.get("K"), fem.get("F"))
}

func buildCFD(mesh, Re, scheme) = {
  {"mesh": mesh, "Re": Re, "scheme": scheme, "u": mesh.map { x -> 0 }}
}

func stepCFD(cfd, dt) = {
  local u = cfd.get("u")
  local n = u.listLen()
  local dx = 1.0 / (n - 1)
  local Re = cfd.get("Re")
  local u_new = copyList(u)
  for (i: 2 .. n-1) {
    local conv = u[i] * (u[i] - u[i-1]) / dx
    local diff = (u[i+1] - 2*u[i] + u[i-1]) / (Re * dx^2)
    u_new[i] = u[i] + dt * (-conv + diff)
  }
  cfd.set("u", u_new)
  cfd
}

func convergenceCFD(cfd) = {
  local u = cfd.get("u")
  local mean_u = u.reduce(0) { x, s -> x + s } / u.listLen()
  abs(mean_u)
}

println(stepCFD(cfd, 0.01))

func rk4_step(ode_fn, y, t, dt) = {
  local k1 = -y
  local k2 = -(y + 0.5*dt*k1)
  local k3 = -(y + 0.5*dt*k2)
  local k4 = -(y + dt*k3)
  y + dt/6 * (k1 + 2*k2 + 2*k3 + k4)
}

func solveODE(ode_fn, y0, t0, tf, dt, method) = {
  local t = t0
  local y = y0
  local ts = [t0]
  local ys = [y0]
  while (t < tf) {
    if (method === "euler") {
      y = y + dt * (-y)
    } else if (method === "rk4") {
      y = rk4_step(ode_fn, y, t, dt)
    } else if (method === "leapfrog") {
      local v = -y * dt / 2
      y = y + v * dt
      v = v - y * dt / 2
      y = y + v
    }
    t = t + dt
    ts.add(t)
    ys.add(y)
  }
  {"ts": ts, "ys": ys}
}

println(solveODE("exp_decay", 1.0, 0, 1.0, 0.1, "rk4"))

func optimise(f, x0, method, iters) = {
  local x = x0
  if (method === "gd") {
    local lr = 0.01
    for (i: 1 .. iters) {
      local grad = 2 * x
      x = x - lr * grad
    }
  } else if (method === "adam") {
    local lr = 0.01
    local beta1 = 0.9
    local beta2 = 0.999
    local eps = 1e-8
    local m_adam = 0
    local v_adam = 0
    for (i: 1 .. iters) {
      local g = 2 * x
      m_adam = beta1 * m_adam + (1 - beta1) * g
      v_adam = beta2 * v_adam + (1 - beta2) * g * g
      local m_hat = m_adam / (1 - beta1 ^ i)
      local v_hat = v_adam / (1 - beta2 ^ i)
      x = x - lr * m_hat / (sqrt(v_hat) + eps)
    }
  } else {
    local step = 1.0
    for (i: 1 .. iters) {
      local fx = x^2
      local x_try = x + (randFloat() - 0.5) * step
      if (x_try^2 < fx) {
        x = x_try
      }
      step = step * 0.995
    }
  }
  {"x_opt": x, "f_opt": x^2}
}

println(optimise("quadratic", 5.0, "adam", 200))

func fft_recursive(signal) = {
  local n = signal.listLen()
  if (n == 1) {
    [[signal[1], 0]]
  } else {
    local even = []
    local odd = []
    for (i: 1 .. n) {
      if (i % 2 == 1) {
        even.add(signal[i])
      } else {
        odd.add(signal[i])
      }
    }
    local E = fft_recursive(even)
    local O = fft_recursive(odd)
    local result = []
    for (i: 1 .. n) {
      result.add([0, 0])
    }
    local half = n / 2
    for (k: 1 .. half) {
      local angle = -2 * 3.14159265 * (k-1) / n
      local wr = cos(angle)
      local wi = sin(angle)
      local ore = wr * O[k][1] - wi * O[k][2]
      local oim = wr * O[k][2] + wi * O[k][1]
      result[k][1] = E[k][1] + ore
      result[k][2] = E[k][2] + oim
      result[k + half][1] = E[k][1] - ore
      result[k + half][2] = E[k][2] - oim
    }
    result
  }
}

println(fft_recursive([1, 1, 1, 1, 0, 0, 0, 0]))
