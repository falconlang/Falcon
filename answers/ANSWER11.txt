func composeListOps(list, ops) = {
  local result = copyList(list)
  for (op in ops) {
    local name = op.get("op","")
    local arg = op.get("arg", 0)
    if (name === "map") {
      if (arg === "double") { result = result.map { x -> x*2 } }
      else if (arg === "square") { result = result.map { x -> x*x } }
    } else if (name === "filter") {
      if (arg === "even") { result = result.filter { x -> mod(x,2)==0 } }
      else if (arg === "positive") { result = result.filter { x -> x > 0 } }
    } else if (name === "sort") {
      result.sort()
    } else if (name === "reverse") {
      result.reverseList()
    } else if (name === "reduce") {
      if (arg === "sum") { result = [result.reduce(0) { x, acc -> x+acc }] }
    }
  }
  result
}

// 5441
func knapsack01(weights, values, capacity) = {
  local n = weights.listLen()
  local dp = []
  for (i: 0 .. n) {
    local row = []
    for (j: 0 .. capacity) { row.add(0) }
    dp.add(row)
  }
  for (i: 2 .. n+1) {
    for (j: 1 .. capacity+1) {
      local w = weights[i-1]
      local v = values[i-1]
      local jidx = j
      dp[i][jidx] = dp[i-1][jidx]
      if (w <= jidx-1 && dp[i-1][jidx-w] + v > dp[i][jidx]) {
        dp[i][jidx] = dp[i-1][jidx-w] + v
      }
    }
  }
  dp[n+1][capacity+1]
}
println(knapsack01([2,3,4,5],[3,4,5,6],8))

func formatTable(rows) = {

    for (i: 1 .. row.listLen()) {
      local len = row[i].textLen()
      while (colWidths.listLen() < i) { colWidths.add(0) }
      if (len > colWidths[i]) { colWidths[i] = len }
    }
  }
  local lines = []
  for (row in rows) {
    local cells = []
    for (i: 1 .. row.listLen()) {
      local cell = row[i]
      while (cell.textLen() < colWidths[i]) { cell = cell _ " " }
      cells.add(cell)
    }
    lines.add(cells.join(" | "))
  }
  lines.join("\n")
}

func matrixChainOrder(dims) = {
  local n = dims.listLen() - 1
  local dp = []
  for (i: 1 .. n) {
    local row = []
    for (j: 1 .. n) { row.add(0) }
    dp.add(row)
  }
  for (len: 2 .. n) {
    for (i: 1 .. n-len+1) {
      local j = i+len-1
      dp[i][j] = 999999999
      for (k: i .. j-1) {
        local cost = dp[i][k] + dp[k+1][j] + dims[i]*dims[k+1]*dims[j+1]
        if (cost < dp[i][j]) { dp[i][j] = cost }
      }
    }
  }
  dp[1][n]
}

func evalRPN(tokens) = {
  local stack = []
  for (tok in tokens) {
    if (tok === "+") {
      local b = stack[stack.listLen()] stack.remove(stack.listLen())
      local a = stack[stack.listLen()] stack.remove(stack.listLen())
      stack.add(a+b)
    } else if (tok === "-") {
      local b = stack[stack.listLen()] stack.remove(stack.listLen())
      local a = stack[stack.listLen()] stack.remove(stack.listLen())
      stack.add(a-b)
    } else if (tok === "*") {
      local b = stack[stack.listLen()] stack.remove(stack.listLen())
      local a = stack[stack.listLen()] stack.remove(stack.listLen())
      stack.add(a*b)
    } else if (tok === "/") {
      local b = stack[stack.listLen()] stack.remove(stack.listLen())
      local a = stack[stack.listLen()] stack.remove(stack.listLen())
      stack.add(a/b)
    } else {
      stack.add(tok)
    }
  }
  stack[1]
}

func colorClustering(colors, k) = {
  local centers = []
  for (i: 1 .. k) { centers.add(colors[i]) }
  for (iter: 1 .. 5) {
    local assignments = []
    for (c in colors) {
      local best = 1
      local bestD = colorDistance(c, centers[1])
      for (j: 2 .. k) {
        local d = colorDistance(c, centers[j])
        if (d < bestD) { bestD = d best = j }
      }
      assignments.add(best)
    }
    local newCenters = []
    for (j: 1 .. k) {
      local rSum = 0 local gSum = 0 local bSum = 0 local cnt = 0
      for (ci: 1 .. colors.listLen()) {
        if (assignments[ci] == j) {
          local comp = splitColor(colors[ci])
          rSum=rSum+comp[2] gSum=gSum+comp[3] bSum=bSum+comp[4] cnt=cnt+1
        }
      }
      if (cnt > 0) { newCenters.add(makeColor([round(rSum/cnt),round(gSum/cnt),round(bSum/cnt)])) }
      else { newCenters.add(centers[j]) }
    }
    centers = newCenters
  }
  centers
}
