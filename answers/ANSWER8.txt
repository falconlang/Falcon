  local divSum = 0
  for (i: 1 .. n-1) { if (n % i == 0) { divSum = divSum + i } }
  if (divSum > n) {
    local amicable = false
    local divSum2 = 0
    for (i: 1 .. divSum-1) { if (divSum % i == 0) { divSum2 = divSum2 + i } }
    if (divSum2 == n && divSum != n) { amicable = true }
    abundantPairs.set(n, amicable)
  }
}
println(abundantPairs)

// 4165
local triPentList = []
local triPentCount = 0
for (n: 1 .. 1000) {
  local tri = n * (n+1) / 2
  local sqrtTerm = sqrt(1 + 24*tri)
  if (mod(round(sqrtTerm), 1) == 0) {
    local m = round(sqrtTerm)
    if (m * m == 1 + 24*tri && mod(m + 1, 6) == 0) {
      local k = (m + 1) / 6
      local pent = k * (3*k - 1) / 2
      if (pent == tri) {
        triPentList.add(tri)
        triPentCount = triPentCount + 1
      }
    }
  }
}
println(triPentCount)
println(triPentList)

// 3797
func jacobiSymbol(a, n) = {
  if (n == 1) { 1 }
  else {
    local result = 1
    local x = mod(a, n)
    local m = n
    while (x != 0) {
      while (mod(x, 2) == 0) {
        x = floor(x / 2)
        local r = mod(m, 8)
        if (r == 3 || r == 5) { result = -result }
      }
      local tmp = m
      m = x
      x = tmp
      if (mod(x, 4) == 3 && mod(m, 4) == 3) { result = -result }
      x = mod(x, m)
    }
    if (m == 1) { result } else { 0 }
  }
}
println(jacobiSymbol(286, 221))

// 3903
local hashStr = "hello"
local hashVal = 0
local hashAlpha = "abcdefghijklmnopqrstuvwxyzABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789 !\"#$%&'()*+,-./:;<=>?@[\\]^_`{|}~"
for (i: 1 .. hashStr.textLen()) {
  local ch = hashStr.segment(i,1)
  local code = 0
  for (j: 1 .. hashAlpha.textLen()) {
    if (hashAlpha.segment(j,1) === ch) { code = j + 31 }
  }
  hashVal = hashVal + code
}
println(mod(hashVal, 97))
