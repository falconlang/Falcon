func stringPermutations(s) = {
  local results = []
  func permute(current, remaining) {
    if (remaining.textLen() == 0) {
      if (!results.containsItem(current)) {
        results.add(current)
      }
    } else {
      for (i: 1 .. remaining.textLen()) {
        local ch = remaining.segment(i, 1)
        local rest = remaining.segment(1, i-1) _ remaining.segment(i+1, remaining.textLen()-i)
        permute(current _ ch, rest)
      }
    }
  }
  permute("", s)
  results.sort()
  results
}
println(stringPermutations("abc"))

func substringSearch(text, patterns) = {
  local result = {}
  for (pattern in patterns) {
    local positions = []
    local plen = pattern.textLen()
    local tlen = text.textLen()
    for (i: 1 .. tlen - plen + 1) {
      if (text.segment(i, plen) === pattern) {
        positions.add(i)
      }
    }
    result.set(pattern, positions)
  }
  result
}

func kmpSearchList(text_chars, pattern_chars) = {
  local m = pattern_chars.listLen()
  local n = text_chars.listLen()
  local fail = []
  for (i: 1 .. m) { fail.add(0) }
  local k = 0
  for (i: 2 .. m) {
    while (k > 0 && pattern_chars[k+1] !== pattern_chars[i]) {
      k = fail[k]
    }
    if (pattern_chars[k+1] === pattern_chars[i]) { k = k + 1 }
    fail[i] = k
  }
  local matches = []
  k = 0
  for (i: 1 .. n) {
    while (k > 0 && pattern_chars[k+1] !== text_chars[i]) {
      k = fail[k]
    }
    if (pattern_chars[k+1] === text_chars[i]) { k = k + 1 }
    if (k == m) {
      matches.add(i - m + 1)
      k = fail[k]
    }
  }
  matches
}
println(kmpSearchList("abcabcabc".split(""), "abc".split("")))

func generateFromMarkov(matrix, start, length) = {
  local result = start
  local cur = start
  for (i: 2 .. length) {
    local row = matrix.get(cur, {})
    local bestProb = -1
    local nextChar = cur
    for (ch, prob in row) {
      if (prob > bestProb) {
        bestProb = prob

func nextPermutation(s) = {
  local chars = []
  for (i: 1 .. s.textLen()) { chars.add(s.segment(i, 1)) }
  local n = chars.listLen()
  local i = n - 1
  while (i >= 1 && chars[i] >= chars[i+1]) {
    i = i - 1
  }
  if (i < 1) {
    chars.reverseList()
  } else {
    local j = n
    while (chars[j] <= chars[i]) { j = j - 1 }
    local tmp = chars[i]
    chars[i] = chars[j]
    chars[j] = tmp
    local suffix = []
    for (k: i+1 .. n) { suffix.add(chars[k]) }
    suffix.reverseList()
    for (k: 1 .. suffix.listLen()) { chars[i+k] = suffix[k] }
  }
  chars.join("")
}

func longestPalindrome(s) = {
  local t = "#"
  for (i: 1 .. s.textLen()) {
    t = t _ s.segment(i, 1) _ "#"
  }
  local n = t.textLen()
  local p = []
  for (i: 1 .. n) { p.add(0) }
  local c = 1
  local r = 1
  for (i: 2 .. n) {
    local mirror = 2 * c - i
    if (i < r && mirror >= 1) {
      p[i] = if (r - i < p[mirror]) r - i else p[mirror]
    }
    local a = i - (p[i] + 1)
    local b = i + (p[i] + 1)
    while (a >= 1 && b <= n && t.segment(a, 1) === t.segment(b, 1)) {
      p[i] = p[i] + 1
      a = a - 1
      b = b + 1
    }
    if (i + p[i] > r) {
      c = i
      r = i + p[i]
    }
  }
  local maxLen = 0
  local center = 1
  for (i: 1 .. n) {
    if (p[i] > maxLen) {
      maxLen = p[i]
      center = i
    }
  }
  local start = floor((center - maxLen) / 2) + 1
  s.segment(start, maxLen)
}
println(longestPalindrome("babad"))
println(longestPalindrome("cbbd"))
println(longestPalindrome("racecar"))

func jaroWinkler(s1, s2) = {
  local l1 = s1.textLen()
  local l2 = s2.textLen()
  if (l1 == 0 && l2 == 0) {
    1
  } else {
    local matchWindow = floor(if (l1 > l2) l1 else l2) / 2 - 1
    local s1matched = []
    local s2matched = []
    for (i: 1 .. l1) { s1matched.add(false) }
    for (i: 1 .. l2) { s2matched.add(false) }
    local matches = 0
    for (i: 1 .. l1) {
      local start = if (i - matchWindow > 1) round(i - matchWindow) else 1
      local end = if (i + matchWindow < l2) round(i + matchWindow) else l2
      for (j: start .. end) {
        if (!s2matched[j] && s1.segment(i,1) === s2.segment(j,1)) {
          s1matched[i] = true
          s2matched[j] = true
          matches = matches + 1
          j = end + 1
        }
      }
    }
    if (matches == 0) {
      0
    } else {
      local trans = 0
      local k = 1
      for (i: 1 .. l1) {
        if (s1matched[i]) {
          while (!s2matched[k]) { k = k + 1 }
          if (s1.segment(i,1) !== s2.segment(k,1)) { trans = trans + 1 }
          k = k + 1
        }
      }
      local jaro = (matches/l1 + matches/l2 + (matches - trans/2)/matches) / 3
      local prefix = 0
      for (i: 1 .. 4) {
        if (i <= l1 && i <= l2 && s1.segment(i,1) === s2.segment(i,1)) {
          prefix = prefix + 1
        }
      }
      jaro + prefix * 0.1 * (1 - jaro)
    }
  }
}
func findMostSimilar(word, candidates) = {
  local best = -1
  local bestWord = ""
  for (c in candidates) {
    local score = jaroWinkler(word, c)
    if (score > best) {
      best = score
      bestWord = c
    }
  }
  bestWord
}
println(jaroWinkler("MARTHA", "MARHTA"))
println(findMostSimilar("helo", ["hello","world","help","hero"]))

func wordLadder(start, end, vocabulary) = {
  local queue = [[start, [start]]]
  local visited = [start]
  while (queue.listLen() > 0) {
    local cur = queue[1]
    queue.remove(1)
    local word = cur[1]
    local path = cur[2]
    if (word === end) {
      path
    } else {
      local vocab = copyList(vocabulary)
      for (candidate in vocab) {
        if (!visited.containsItem(candidate)) {
          local diffs = 0
          if (candidate.textLen() == word.textLen()) {
            for (i: 1 .. word.textLen()) {
              if (word.segment(i,1) !== candidate.segment(i,1)) {
                diffs = diffs + 1
              }
            }
            if (diffs == 1) {
              visited.add(candidate)
              local newPath = copyList(path)
              newPath.add(candidate)
              queue.add([candidate, newPath])
            }
          }
        }
      }
      if (queue.listLen() == 0) { [] }
    }
  }
}
println(wordLadder("hit", "cog", ["hot","dot","dog","lot","log","cog"]))

func decompressString(s) = {
  local stack = []
  local currentStr = ""
  local currentNum = 0
  for (i: 1 .. s.textLen()) {
    local ch = s.segment(i, 1)
    if (ch ? number) {
      currentNum = currentNum * 10 + ch
    } else if (ch === "[") {
      stack.add([currentNum, currentStr])
      currentStr = ""
      currentNum = 0
    } else if (ch === "]") {
      local top = stack[stack.listLen()]
      stack.remove(stack.listLen())
      local times = top[1]
      local prev = top[2]
      local repeated = ""
      for (k: 1 .. times) { repeated = repeated _ currentStr }
      currentStr = prev _ repeated
    } else {
      currentStr = currentStr _ ch
    }
  }
  currentStr
}

func detectAnomalousWords(text, languageModel) = {
  local words = text.splitAtSpaces()
  local result = []
  local threshold = 0.01
  for (i: 2 .. words.listLen()) {
    local prev = words[i-1]
    local word = words[i]
    local bigrams = languageModel.get("bigrams", {})
    local unigrams = languageModel.get("unigrams", {})
    local total = languageModel.get("total", 1)
    local pPrev = unigrams.get(prev, 0) / total
    local pWord = unigrams.get(word, 0) / total
    local pBigram = bigrams.get(prev _ " " _ word, 0) / total
    local prob = if (pPrev > 0) pBigram / pPrev else 0
    if (prob < threshold) {
      result.add({"word": word, "position": i, "prob": prob})
    }
  }
  result
}

func autocomplete(trie, prefix) = {
  local node = trie
  for (i: 1 .. prefix.textLen()) {
    local ch = prefix.segment(i, 1)
    if (!node.containsKey(ch)) {
      []
    }
    node = node.get(ch, {})
  }
  local results = []
  func collect(n, current) {
    if (n.containsKey("$")) { results.add(current) }
    for (key, val in n) {
      if (key !== "$") {
        collect(val, current _ key)
      }
    }
  }
  collect(node, prefix)
  results
}

  local bigrams = {}

    unigrams.set(w, unigrams.get(w, 0) + 1)

    local pb = unigrams.get(b, 1) / n
    if (!pmi.containsKey(a)) { pmi.set(a, {}) }
    pmi.get(a, {}).set(b, log(pab / (pa * pb)))

}

func longestPalindromicSubstring(s) = {

        maxLen = r - l + 1
        start = l

      r = r + 1

func boyerMooreHeuristic(text, pattern) = {
  local plen = pattern.textLen()
  local tlen = text.textLen()
  local badChar = {}
  for (i: 1 .. plen) {
    badChar.set(pattern.segment(i,1), i)
  }
  local matches = []
  local s = 1
  while (s <= tlen - plen + 1) {
    local j = plen
    while (j >= 1 && pattern.segment(j,1) === text.segment(s+j-1,1)) {
      j = j - 1
    }
    if (j == 0) {
      matches.add(s)
      s = s + 1
    } else {
      local bc = badChar.get(text.segment(s+j-1,1), 0)
      local shift = j - bc
      s = s + if (shift > 1) shift else 1
    }
  }
  matches
}
println(boyerMooreHeuristic("ABCABCABC","ABC"))

func jaroSimilarity(s1, s2) = {
  local l1 = s1.textLen()
  local l2 = s2.textLen()
  local matchWindow = floor((if (l1 > l2) l1 else l2) / 2) - 1
  local s1matched = []
  local s2matched = []
  for (i: 1 .. l1) { s1matched.add(false) }
  for (i: 1 .. l2) { s2matched.add(false) }
  local matches = 0
  for (i: 1 .. l1) {
    local lo = if (i - matchWindow > 1) i - matchWindow else 1
    local hi = if (i + matchWindow < l2) i + matchWindow else l2
    for (j: lo .. hi) {
      if (!s2matched[j] && !s1matched[i] && s1.segment(i,1) === s2.segment(j,1)) {
        s1matched[i] = true
        s2matched[j] = true
        matches = matches + 1
      }
    }
  }
  if (matches == 0) {
    0
  } else {
    local trans = 0
    local k = 1
    for (i: 1 .. l1) {
      if (s1matched[i]) {
        while (k <= l2 && !s2matched[k]) { k = k + 1 }
        if (k <= l2 && s1.segment(i,1) !== s2.segment(k,1)) { trans = trans + 1 }
        k = k + 1
      }
    }
    (matches/l1 + matches/l2 + (matches - trans/2)/matches) / 3
  }
}
println(jaroSimilarity("MARTHA","MARHTA"))

func extractMarkdown(text) = {
  local result = text
  local i = 1
  local out = ""
  while (i <= result.textLen()) {
    if (i <= result.textLen() - 1 && result.segment(i,2) === "**") {
      local end = i + 2
      local content = ""
      while (end <= result.textLen() - 1 && result.segment(end,2) !== "**") {
        content = content _ result.segment(end,1)
        end = end + 1
      }
      out = out _ "[BOLD]" _ content _ "[/BOLD]"
      i = end + 2
    } else if (result.segment(i,1) === "_") {
      local end = i + 1
      local content = ""
      while (end <= result.textLen() && result.segment(end,1) !== "_") {
        content = content _ result.segment(end,1)
        end = end + 1
      }
      out = out _ "[ITALIC]" _ content _ "[/ITALIC]"
      i = end + 1
    } else if (i == 1 && result.segment(i,2) === "# ") {
      out = "[H1]" _ result.segment(3, result.textLen()-2) _ "[/H1]"
      i = result.textLen() + 1
    } else {
      out = out _ result.segment(i,1)
      i = i + 1
    }
  }
  out
}

func buildCharAlignmentMatrix(s1, s2) = {
  local m = s1.textLen()
  local n = s2.textLen()
  local match = 1
  local mismatch = -1
  local gap = -2
  local dp = []
  for (i: 0 .. m) {
    local row = []
    for (j: 0 .. n) { row.add(0) }
    dp.add(row)
  }
  for (i: 0 .. m) { dp[i][1] = i * gap }
  for (j: 0 .. n) { dp[1][j] = j * gap }
  for (i: 2 .. m + 1) {
    for (j: 2 .. n + 1) {
      local score = if (s1.segment(i-1,1) === s2.segment(j-1,1)) match else mismatch
      local diag = dp[i-1][j-1] + score
      local up = dp[i-1][j] + gap
      local left = dp[i][j-1] + gap
      dp[i][j] = if (diag > up && diag > left) diag else if (up > left) up else left
    }
  }
  dp
}

func editDistance(s1, s2) = {
  local m = s1.textLen()
  local n = s2.textLen()
  local dp = []
  for (i: 0 .. m) {
    local row = []
    for (j: 0 .. n) { row.add(0) }
    dp.add(row)
  }
  for (i: 0 .. m) { dp[i][1] = i }
  for (j: 0 .. n) { dp[1][j] = j }
  for (i: 2 .. m + 1) {
    for (j: 2 .. n + 1) {
      if (s1.segment(i-1,1) === s2.segment(j-1,1)) {
        dp[i][j] = dp[i-1][j-1]
      } else {
        local a = dp[i-1][j] + 1
        local b = dp[i][j-1] + 1
        local c = dp[i-1][j-1] + 1
        dp[i][j] = if (a < b && a < c) a else if (b < c) b else c
      }
    }
  }
  dp[m+1][n+1]
}
func lcsLength(s1, s2) = {
  local m = s1.textLen()
  local n = s2.textLen()
  local dp = []
  for (i: 0 .. m) {
    local row = []
    for (j: 0 .. n) { row.add(0) }
    dp.add(row)
  }
  for (i: 1 .. m) {
    for (j: 1 .. n) {
      if (s1.segment(i,1) === s2.segment(j,1)) {
        dp[i][j] = dp[i-1][j-1] + 1
      } else {
        dp[i][j] = if (dp[i-1][j] > dp[i][j-1]) dp[i-1][j] else dp[i][j-1]
      }
    }
  }
  dp[m][n]
}
func commonBigrams(s1, s2) = {
  local bg1 = {}
  local bg2 = {}
  for (i: 1 .. s1.textLen()-1) { local g = s1.segment(i,2) bg1.set(g, bg1.get(g,0)+1) }
  for (i: 1 .. s2.textLen()-1) { local g = s2.segment(i,2) bg2.set(g, bg2.get(g,0)+1) }
  local count = 0
  for (g, c in bg1) {
    if (bg2.containsKey(g)) { count = count + if (c < bg2.get(g,0)) c else bg2.get(g,0) }
  }
  count
}

  local wordInText = {}
  local n = texts.listLen()
  for (text in texts) {
    local words = text.splitAtSpaces()

        seen.add(wl)
        wordInText.set(wl, wordInText.get(wl, 0) + 1)
      }
    }
  }
  local result = {}
  for (w, cnt in wordInText) {
    local frac = cnt / n
    if (frac > threshold) { result.set(w, frac) }
  }
  result
}

func buildHuffmanCodes(freqDict) = {
  local heap = []
  for (ch, freq in freqDict) {
    heap.add([freq, ch, {}])
  }
  heap.sort()
  while (heap.listLen() > 1) {
    local a = heap[1]
    heap.remove(1)
    local b = heap[1]
    heap.remove(1)
    local merged = [a[1] + b[1], a[2] _ b[2], {"0": a, "1": b}]
    heap.add(merged)
    heap.sort()
  }
  local codes = {}
  func assign(node, prefix) {
    if (node[3].dictLen() == 0) {
      codes.set(node[2], prefix)
    } else {
      assign(node[3].get("0", [0,"",{}]), prefix _ "0")
      assign(node[3].get("1", [0,"",{}]), prefix _ "1")
    }
  }
  if (heap.listLen() > 0) { assign(heap[1], "") }
  codes
}
println(buildHuffmanCodes({"a":5,"b":3,"c":1,"d":2}))

func sentenceSegment(text) = {
  local sentences = []
  local current = ""
  local i = 1
  while (i <= text.textLen()) {
    local ch = text.segment(i,1)
    current = current _ ch
    if (ch === "." || ch === "!" || ch === "?") {
      local isAbbrev = false
      if (ch === "." && i >= 2) {
        local prev = text.segment(i-1,1)
        local uppers = "ABCDEFGHIJKLMNOPQRSTUVWXYZ"
        if (uppers.contains(prev) && (i < 3 || text.segment(i-2,1) === " ")) {
          isAbbrev = true
        }
      }
      if (!isAbbrev) {
        sentences.add(current.trim())
        current = ""
      }
    }
    i = i + 1
  }
  if (current.trim().textLen() > 0) { sentences.add(current.trim()) }
  sentences
}
println(sentenceSegment("Hello world. How are you? I am fine! Dr. Smith said hello."))
