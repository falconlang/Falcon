// 7192
func computeGiniSplit(leftCounts, rightCounts) = {
  local leftTotal = leftCounts.reduce(0){x,acc->acc+x}
  local rightTotal = rightCounts.reduce(0){x,acc->acc+x}
  local total = leftTotal + rightTotal
  local giniLeft = 1
  for (c in leftCounts) { if (leftTotal>0) { giniLeft = giniLeft - (c/leftTotal)^2 } }
  local giniRight = 1
  for (c in rightCounts) { if (rightTotal>0) { giniRight = giniRight - (c/rightTotal)^2 } }
  (leftTotal/total)*giniLeft + (rightTotal/total)*giniRight
}
println(formatDecimal(computeGiniSplit([30,10],[5,55]),4))
